const clamp01 = (x) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a, b, x) => { const t = clamp01((x - a) / (b - a)); return t * t * (3 - 2 * t); };
const h11 = (p) => { let x = Math.sin(p * 127.1) * 43758.5453; return x - Math.floor(x); };
const easeOutExpo = (x) => (x >= 1 ? 1 : 1 - Math.pow(2, -9 * x));
const easeOutCubic = (x) => 1 - Math.pow(1 - clamp01(x), 3);

// the four envelopes the whole shot hangs off. Everything else is a product of these.
// build 0→1.5, hold, collapse 3.5→5.0 (height retracts from the top, width narrows)
const colEnv = (t) => (0.20 + 0.80 * ss(0, 1.5, t)) * (1 - ss(3.85, 4.95, t));
const colHeight = (t) => 1 - 0.97 * ss(3.60, T_OUT1, t);
const colWidth = (t) => 1 - 0.72 * ss(T_OUT0, 4.90, t);
// arcs: grow in, densest right before the eruption, thinned during the sustain, gone by 4.0
const arcEnv = (t) => ss(0.05, 1.10, t) * (1 + 0.85 * ss(1.05, 1.48, t))
  * (1 - 0.62 * ss(1.95, 2.60, t)) * (1 - ss(3.35, 4.00, t));

const GLSL = `
#define TAU 6.28318530718
float h11(float p){ p = fract(p*0.1031); p *= p+33.33; p *= p+p; return fract(p); }
float h21(vec2 p){ vec3 q = fract(vec3(p.xyx)*0.1031); q += dot(q,q.yzx+33.33); return fract((q.x+q.y)*q.z); }
float ssm(float a, float b, float x){ float t = clamp((x-a)/(b-a),0.0,1.0); return t*t*(3.0-2.0*t); }
float vnoise(vec2 p){
  vec2 i = floor(p), f = fract(p); f = f*f*f*(f*(f*6.0-15.0)+10.0);
  return mix(mix(h21(i), h21(i+vec2(1,0)), f.x), mix(h21(i+vec2(0,1)), h21(i+vec2(1,1)), f.x), f.y);
}
const mat2 M2 = mat2(0.80, 0.60, -0.60, 0.80);
float fbm3(vec2 p){ float s = 0.0, a = 0.5;
  for (int i = 0; i < 3; i++){ s += a*vnoise(p); p = M2*p*2.03 + vec2(11.3,7.1); a *= 0.5; }
  return s/0.875; }
// the palette, in linear light. Only the inner core and the flare pass 1.0.
const vec3 C_CORE = vec3(1.000, 0.871, 0.445);  // #fff0b0 white-gold core
const vec3 C_GOLD = vec3(1.000, 0.434, 0.014);  // #ffb020 body, low in the column
const vec3 C_ORN  = vec3(1.000, 0.195, 0.005);  // #ff7a10 body, high in the column
const vec3 C_ARC  = vec3(0.738, 0.905, 1.000);  // #dff4ff electrical core
// the three streak hues. Hashed per streak, so the burst is a spread, not one colour.
const vec3 C_ST_A = vec3(1.000, 0.254, 0.022);  // #ff8a2a saturated orange
const vec3 C_ST_B = vec3(1.000, 0.069, 0.254);  // #ff4a8a hot pink
const vec3 C_ST_C = vec3(1.000, 0.644, 0.195);  // #ffd27a pale gold
`;

// screen-aligned quad anchored at the object's own origin: every billboarded layer
// offsets in VIEW space so nothing shears when the column tilts away from the camera
function quadXY() {
  const g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute([-0.5, -0.5, 0, 0.5, -0.5, 0, -0.5, 0.5, 0, 0.5, 0.5, 0], 3));
  g.setAttribute("uv", new THREE.Float32BufferAttribute([0, 0, 1, 0, 0, 1, 1, 1], 2));
  g.setIndex([0, 1, 2, 1, 3, 2]);
  return g;
}
function billboard(frag, uniforms, order, blend, extraVs = "") {
  const m = new THREE.Mesh(quadXY(), new THREE.ShaderMaterial({
    uniforms,
    vertexShader: `${GLSL}
      uniform float uW, uH; varying vec2 vUv; ${extraVs}
      void main(){
        vUv = uv;
        vec4 mv = modelViewMatrix * vec4(0.0, 0.0, 0.0, 1.0);
        mv.xy += vec2(position.x*uW, position.y*uH);
        gl_Position = projectionMatrix * mv;
      }`,
    fragmentShader: frag,
    transparent: true,
    depthWrite: false,
    depthTest: false,
    blending: blend,
  }));
  m.renderOrder = order;
  m.frustumCulled = false;
  return m;
}

    void main(){ vP = position.xy; gl_Position = projectionMatrix*modelViewMatrix*vec4(position,1.0); }`,

// ---------------------------------------------------------------- column body
// a screen-aligned slab BEHIND the tapered shell. The shell alone reads as a tube;
// the slab is what gives the column its readable body width from any angle.
const slabU = { uW: { value: 1 }, uH: { value: 1 }, uI: { value: 0 }, uT: { value: 0 }, uHt: { value: 1 } };
const slab = billboard(`${GLSL}
  varying vec2 vUv; uniform float uI, uT, uHt;
  void main(){
    float x = (vUv.x-0.5)*2.0;
    float h = vUv.y;
    float w = mix(1.00, 0.74, h);                    // narrows with height, like the shell
    float q = abs(x)/w;
    // hard tiers, not a gaussian: a soft-edged slab reads as fog, a tiered one as a bar
    float inner = 1.0 - ssm(0.20, 0.30, q);
    float body  = 1.0 - ssm(0.44, 0.56, q);
    float outer = 1.0 - ssm(0.70, 0.94, q);
    float halo  = exp(-q*q*0.75)*0.16;
    float streak = 0.62 + 0.85*pow(vnoise(vec2(x*4.5, h*7.0 - uT*0.85)), 1.3);
    // the slab carries the same segment ladder, so the banding reads across the body
    float seg = (h - uT*0.055)*7.0;
    float ring = 1.0 - ssm(0.10, 0.22, abs(fract(seg) - 0.10));
    streak *= (0.80 + 0.62*ring) * (0.80 + 0.40*h11(floor(seg)*3.17 + 1.7));
    float top = 1.0 - ssm(0.86, 1.02, h);            // soft top, never a flat cap
    float base = ssm(0.0, 0.07, h);
    vec3 c = mix(C_GOLD, C_ORN, ssm(0.05, 0.95, h));
    c = mix(C_ORN, c, body*0.85 + 0.15);             // the outer tier stays orange
    c = mix(c, mix(C_GOLD, C_CORE, 0.40), inner*0.45);
    float a = (outer*0.22 + body*0.34 + inner*0.26 + halo) * streak * top * base;
    gl_FragColor = vec4(min(c*a*uI*0.52, vec3(0.90)), 1.0);
  }`, slabU, 2, THREE.AdditiveBlending);
slab.position.y = COL_Y;

// the tapered shell: r 0.32 at the base → 0.22 at the top, gold → orange by HEIGHT,
// with two panning streak frequencies, a fresnel rim and a soft top.
const shellGeo = new THREE.CylinderGeometry(0.24, 0.34, COL_H, 48, 1, true);
shellGeo.translate(0, COL_H * 0.5, 0);
const shellU = { uT: { value: 0 }, uI: { value: 0 }, uHt: { value: 1 }, uWd: { value: 1 } };
const shellVS = `${GLSL}
  uniform float uHt, uWd; varying vec2 vUv; varying float vH; varying vec3 vN, vV;
  void main(){
    vUv = uv; vH = position.y/${COL_H.toFixed(2)};
    vec3 p = vec3(position.x*uWd, position.y*uHt, position.z*uWd);
    vN = normalize(normalMatrix*vec3(normal.x, 0.0, normal.z));
    vec4 mv = modelViewMatrix*vec4(p,1.0);
    vV = normalize(-mv.xyz);
    gl_Position = projectionMatrix*mv;
  }`;
const shellMat = new THREE.ShaderMaterial({
  uniforms: shellU,
  vertexShader: shellVS,
  fragmentShader: `${GLSL}
    varying vec2 vUv; varying float vH; varying vec3 vN, vV; uniform float uT, uI, uHt;
    void main(){
      float h = vH;
      // two frequencies panning UP: the slow one is the band, the fast one the filament
      float n1 = vnoise(vec2(vUv.x*8.0, h*4.5 - uT*0.85));
      float n2 = vnoise(vec2(vUv.x*19.0+3.1, h*12.0 - uT*1.75));
      float streak = 0.30 + 1.05*pow(n1*0.62 + n2*0.38, 1.5);   // contrast: filaments, not fog
      // 7 machine-like segments: a 0.05-tall bright ring, a darker gap behind it,
      // and a stepped brightness hashed per segment. floor(seg) makes it discrete.
      float seg = (h - uT*0.055)*7.0;
      float f = fract(seg);
      float ring = 1.0 - ssm(0.10, 0.22, abs(f - 0.10));
      float gap  = ssm(0.26, 0.44, f) * (1.0 - ssm(0.78, 0.94, f));
      float sstep = 0.78 + 0.42*h11(floor(seg)*3.17 + 1.7);
      streak *= (0.70 + 1.05*ring - 0.30*gap) * sstep;
      float fres = pow(1.0 - abs(dot(normalize(vN), normalize(vV))), 1.6);
      float top  = 1.0 - ssm(0.84, 1.03, h/max(uHt,0.001));  // retracts from the top
      float base = ssm(0.0, 0.09, h);
      vec3 c = mix(C_GOLD, C_ORN, ssm(0.08, 0.92, h));
      c = mix(c, C_CORE, fres*fres*0.16*streak);
      float a = (0.72 + 0.48*fres) * streak * top * base;
      gl_FragColor = vec4(min(c*a*uI*0.46, vec3(0.90)), 1.0);
    }`,
  transparent: true,
  depthWrite: false,
  blending: THREE.AdditiveBlending,
  side: THREE.DoubleSide,
});
const shell = new THREE.Mesh(shellGeo, shellMat);
shell.position.y = COL_Y;
shell.renderOrder = 3;
scene.add(shell);

// ---------------------------------------------------------------- arcs
// helical polylines wrapping the column, drawn as CAMERA-FACING ribbons: the strip
// carries only (u, side) and the world path is swept in the vertex shader, so the
// ribbon always presents its width to the camera however the arc twists.
const ARC_SEG = 26;
const arcGeo = (() => {
  const pos = [], idx = [];
  for (let i = 0; i <= ARC_SEG; i++) { const u = i / ARC_SEG; pos.push(u, -1, 0, u, 1, 0); }
  for (let i = 0; i < ARC_SEG; i++) { const a = i * 2; idx.push(a, a + 2, a + 1, a + 1, a + 2, a + 3); }
  const g = new THREE.InstancedBufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setIndex(idx);
  const seed = new Float32Array(N_ARC);
  for (let i = 0; i < N_ARC; i++) seed[i] = i * 3.77 + 0.91;
  g.setAttribute("aSeed", new THREE.InstancedBufferAttribute(seed, 1));
  g.instanceCount = N_ARC;
  return g;
})();
const arcU = { uT: { value: 0 }, uI: { value: 0 }, uHt: { value: 1 }, uWd: { value: 1 } };
const arcs = new THREE.Mesh(arcGeo, new THREE.ShaderMaterial({
  uniforms: arcU,
  vertexShader: `${GLSL}
    attribute float aSeed; uniform float uT, uI, uHt, uWd;
    varying vec2 vUv; varying float vA;
    // one blink's worth of helix: radius, pitch, span and phase are re-hashed on
    // every blink (k), so no two flashes trace the same wire
    vec3 pt(float u, float s, float k){
      float r  = (0.26 + 0.26*h11(s*3.11 + k*1.7)) * uWd;
      float pv = 0.40 + 1.40*h11(s*7.73 + k*2.3);
      float y0 = 0.10 + 2.30*h11(s*5.31 + k*3.1);
      float ln = 0.55 + 1.70*h11(s*11.1 + k*0.7);
      float ph = h11(s*13.7 + k*4.9)*TAU;
      float y  = (y0 + u*ln) * uHt;
      float a  = ph + u*pv*TAU;
      // fbm jitter along the path: the wire wanders instead of lying on the cylinder
      // folded noise: |n-0.5| has corners, so the wire kinks instead of curling
      float j1 = abs(fbm3(vec2(u*9.0, s*3.1 + k*0.31)) - 0.5)*2.0 - 0.5;
      float j2 = abs(fbm3(vec2(u*22.0+7.0, s*5.7 + k*0.73)) - 0.5)*2.0 - 0.5;
      float j3 = fbm3(vec2(u*38.0+3.0, s*7.9 + k*1.13)) - 0.5;
      r *= 1.0 + 0.70*j1 + 0.22*j3;
      a += 1.45*j2 + 0.40*j3;
      return vec3(cos(a)*r, y, sin(a)*r);
    }
    void main(){
      float s = aSeed, u = position.x, side = position.y;
      // per-arc stepped window: on 60–140 ms out of a 200–500 ms period, and some
      // periods are skipped outright, so the set never settles into a rhythm
      float per = 0.18 + 0.20*h11(s*1.71);
      float onT = 0.07 + 0.08*h11(s*2.93);
      float off = h11(s*4.37)*per;
      float k   = floor((uT - off)/per);
      float loc = (uT - off) - k*per;
      float live = step(loc, onT) * step(0.26, h11(k*1.373 + s*9.11));
      float w = sin(3.14159265*clamp(loc/max(onT,1e-4), 0.0, 1.0));
      vA = live * w * (0.55 + 0.45*h11(k*2.11 + s*5.5));
      vUv = vec2(u, side*0.5+0.5);
      vec3 a3 = pt(u, s, k);
      vec3 b3 = pt(min(u+0.035,1.0) + (u>0.965 ? -0.070 : 0.0), s, k);
      vec4 mv = modelViewMatrix*vec4(a3,1.0);
      vec4 mb = modelViewMatrix*vec4(b3,1.0);
      vec2 d = mb.xy - mv.xy;
      vec2 pp = length(d) > 1e-5 ? normalize(vec2(-d.y, d.x)) : vec2(1.0,0.0);
      float wid = (0.020 + 0.016*h11(s*17.3)) * (0.55 + 0.45*sin(3.14159*u));
      mv.xy += pp*side*wid;
      gl_Position = projectionMatrix*mv;
    }`,
  fragmentShader: `${GLSL}
    varying vec2 vUv; varying float vA; uniform float uI;
    void main(){
      float y = abs(vUv.y-0.5)*2.0;
      float core = 1.0 - ssm(0.10, 0.55, y);   // white-blue filament
      float halo = 1.0 - ssm(0.30, 1.00, y);   // thin gold sheath around it
      float ends = ssm(0.0,0.06,vUv.x)*(1.0-ssm(0.94,1.0,vUv.x));
      vec3 c = C_ARC*core*1.85 + C_GOLD*halo*0.75;
      gl_FragColor = vec4(c*vA*ends*uI, 1.0);
    }`,
  transparent: true,
  depthWrite: false,
  depthTest: false,
  blending: THREE.AdditiveBlending,
  side: THREE.DoubleSide,
}));
arcs.frustumCulled = false;
arcs.position.y = COL_Y;
arcs.renderOrder = 5;
scene.add(arcs);

// ---------------------------------------------------------------- eruption
// the flare sprite: one card, several gaussians. A core, two ghosts, a horizontal
// and a vertical lens streak, and four soft spikes — a lens flare, not a blob.
const flareU = { uW: { value: 1 }, uH: { value: 1 }, uI: { value: 0 }, uT: { value: 0 } };
const flare = billboard(`${GLSL}
  varying vec2 vUv; uniform float uI, uT;
  void main(){
    vec2 q = (vUv-0.5)*2.0;
    float r = length(q);

// ---------------------------------------------------------------- per-frame state
function apply(t) {
  const L = P.layers;
  const env = colEnv(t);
  const ht = colHeight(t);
  const wd = colWidth(t);
  // the eruption briefly overdrives the body, then the sustain holds it just above
  // the pre-flare level so the core stays readable through the flare
  const kick = 1 + 0.55 * ss(T_ERUPT0, 1.85, t) * (1 - ss(2.05, 2.75, t));
  const sustain = 1 + 0.28 * ss(2.15, 2.50, t) * (1 - ss(3.30, 4.10, t));

  shellU.uT.value = t; shellU.uI.value = env * kick; shellU.uHt.value = ht; shellU.uWd.value = wd;
  slabU.uT.value = t; slabU.uI.value = env * kick; slabU.uHt.value = COL_H * ht; slabU.uW.value = 1.45 * wd;
  slabU.uH.value = COL_H * ht;
  slab.position.y = COL_Y + 0.5 * COL_H * ht;
  coreU.uT.value = t; coreU.uI.value = env * kick * sustain * 0.78; coreU.uHt.value = ht; coreU.uWd.value = wd;
  shell.visible = slab.visible = core.visible = L.column && env > 1e-3;
  cylMat.uniforms.uI.value = env;

  const ai = arcEnv(t);
  arcU.uT.value = t; arcU.uI.value = ai; arcU.uHt.value = ht; arcU.uWd.value = wd;
  arcGeo.instanceCount = Math.max(0, Math.round(12 + 6 * ss(0.3, 1.45, t) - 6 * ss(2.0, 2.7, t)));
  arcs.visible = L.arcs && ai > 1e-3;

  // easeOutExpo scale-up over the eruption window, then a slow dim
  const fg = easeOutExpo(clamp01((t - T_ERUPT0) / 0.80));
  const fi = ss(T_ERUPT0, 1.66, t) * (1 - 0.80 * ss(2.55, 4.30, t)) * (1 - ss(4.30, 4.95, t));
  flareU.uT.value = t; flareU.uI.value = fi * 0.56;
  flareU.uW.value = flareU.uH.value = (0.5 + 3.4 * fg) * (1 - 0.45 * ss(3.6, 4.9, t));
  flare.visible = L.flare && fi > 1e-3;

  flashU.uA.value = 0.45 * ss(1.80, 1.90, t) * (1 - ss(1.93, 2.05, t));
  flash.visible = L.flare && flashU.uA.value > 1e-3;

  streakU.uT.value = t;
  streakU.uG.value = easeOutCubic(clamp01((t - 1.52) / 1.70));
  streakU.uI.value = ss(T_ERUPT0, 1.64, t) * (1 - ss(T_ERUPT1, T_OUT0, t) * 0.88) * (1 - ss(3.45, 4.20, t));
  streaks.visible = L.streaks && streakU.uI.value > 1e-3;

  // the ring snaps out over 0.4 s and is gone; nothing about it accumulates
  const rk = clamp01((t - 1.72) / 0.40);
  ringU.uK.value = rk;
  ringU.uA.value = 0.95 * ss(1.72, 1.76, t) * Math.pow(1 - rk, 1.8);
  const rr = 0.5 + 1.7 * (1 - Math.pow(1 - rk, 2.6));
  shockRing.scale.set(rr, rr, 1);
  shockRing.visible = L.flare && ringU.uA.value > 1e-3;

  debrisU.uT.value = t;
  debrisU.uI.value = 1 - ss(3.30, 4.10, t);
  debris.visible = L.sparks && t > T_ERUPT0 && debrisU.uI.value > 1e-3;

  sparkU.uT.value = t;
  sparkU.uI.value = (1 - ss(3.20, 4.00, t));
  sparks.visible = L.sparks && t > T_ERUPT0 && sparkU.uI.value > 1e-3;

  hazeU.uT.value = t;
  hazeU.uI.value = (env * 0.24 + 0.26 * ss(T_ERUPT0, 1.95, t) * (1 - ss(2.3, 4.4, t))) * 0.28;
  haze.visible = L.column && hazeU.uI.value > 1e-3;

  groundMat.uniforms.uT.value = t;
  groundMat.uniforms.uI.value = env * kick * 0.85 + 0.55 * ss(T_ERUPT0, 1.9, t) * (1 - ss(2.1, 3.2, t));
  groundMat.uniforms.uPool.value = L.pool ? 1 : 0;

  bloom.enabled = L.post;
}

let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1e3);
  last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom;
  bloom.threshold = P.threshold;
  apply(P.time);
  composer.render();
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

window.spikeSetTime = (t) => { P.play = false; P.time = t; apply(t); };
window.__spikeReady = true;
