// state = f(t, seed): every layer reads the clock, nothing accumulates, seek == play.

const DUR = 6.0;
const CY = 3.0;             // disc centre height
const R0 = 2.0;             // outer swirl disc radius (diameter 4)
const TILT_X = -0.52;       // disc pitched away from the camera: reads as an oblique disc
const TILT_Y = 0.20;
const N_PUFF = 58;          // cloud billboards on the disc plane
const N_FLECK = 120;
const T_CORE0 = 0.0, T_CORE1 = 0.40;   // core lights first
const T_DISC0 = 0.20, T_DISC1 = 1.00;  // bands fade in, swirl strength rises from 0
const T_OUT0 = 4.80, T_OUT1 = 6.00;    // dissipate

const P = (window.spikeParams = {
  play: true,
  time: 0,
  speed: 1,
  bloom: 0.35,
  threshold: 1.0,
  exposure: 1.0,
  layers: { discs: true, puffs: true, core: true, haze: true, flecks: true, post: true },
});

const clamp01 = (x) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a, b, x) => { const t = clamp01((x - a) / (b - a)); return t * t * (3 - 2 * t); };
const h11 = (p) => { let x = Math.sin(p * 127.1) * 43758.5453; return x - Math.floor(x); };

// the one shared envelope: how hard the polar swirl bites. 0 → the bands are
// straight noise, 1 → a tight log spiral. Reveal winds it up, the fade unwinds it.
const swirlEnv = (t) => ss(T_DISC0, T_DISC1, t) * (1 - ss(T_OUT0, 5.75, t));
// erosion threshold: bands tear apart from the outside in during the dissipate
const erodeEnv = (t) => ss(T_OUT0, 5.80, t);

const canvas = document.getElementById("c");
const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, powerPreference: "high-performance" });
renderer.setPixelRatio(1);
renderer.setSize(1280, 720, false);
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = P.exposure;
renderer.setClearColor(0x000000, 1);

const scene = new THREE.Scene();

// fixed camera looking slightly UP at the suspended disc
const camera = new THREE.PerspectiveCamera(32, 1280 / 720, 0.1, 200);
camera.position.set(0, 1.55, 5.15);
camera.lookAt(0, CY, 0);

const GLSL = `
#define TAU 6.28318530718
float h11(float p){ p = fract(p*0.1031); p *= p+33.33; p *= p+p; return fract(p); }
float h21(vec2 p){ vec3 q = fract(vec3(p.xyx)*0.1031); q += dot(q,q.yzx+33.33); return fract((q.x+q.y)*q.z); }
float ssm(float a, float b, float x){ float t = clamp((x-a)/(b-a),0.0,1.0); return t*t*(3.0-2.0*t); }
vec2 rot2(vec2 p, float a){ float c = cos(a), s = sin(a); return vec2(p.x*c - p.y*s, p.x*s + p.y*c); }
float vnoise(vec2 p){
  vec2 i = floor(p), f = fract(p); f = f*f*f*(f*(f*6.0-15.0)+10.0);
  return mix(mix(h21(i), h21(i+vec2(1,0)), f.x), mix(h21(i+vec2(0,1)), h21(i+vec2(1,1)), f.x), f.y);
}
// 2 octaves is all the band mask needs; the lobe pass adds the cauliflower edge
float fbm2(vec2 p){ return (vnoise(p) + 0.5*vnoise(p*2.07+19.3)) / 1.5; }
// the band mask uses 3 rotated octaves — an unrotated stack lines its lobes up on the grid
const mat2 M2 = mat2(0.80, 0.60, -0.60, 0.80);
float fbm3(vec2 p){ float s = 0.0, a = 0.5;
  for (int i = 0; i < 3; i++){ s += a*vnoise(p); p = M2*p*2.03 + vec2(11.3,7.1); a *= 0.5; }
  return s/0.875; }
// the palette, in linear light. Only the core is allowed past 1.0.
const vec3 C_HI  = vec3(1.000, 0.880, 0.470);  // #fff1b0 bright core
const vec3 C_LIT = vec3(1.000, 0.300, 0.040);  // crest of a band, catching the core
const vec3 C_MID = vec3(0.920, 0.125, 0.020);  // #ffa030 lit band face
const vec3 C_SHA = vec3(0.380, 0.026, 0.009);  // #c0330c band shadow
const vec3 C_HAZ = vec3(0.310, 0.021, 0.011);  // #7a1a10 outer haze
`;

const vortex = new THREE.Group();
vortex.position.set(0, CY, 0);
vortex.rotation.set(TILT_X, TILT_Y, 0);
scene.add(vortex);

// ---------------------------------------------------------------- sky
// a deep red-black gradient behind everything; no ground, the vortex hangs in it
function billboard(size, frag, uniforms, order, blend) {
  const m = new THREE.Mesh(
    new THREE.PlaneGeometry(size, size),
    new THREE.ShaderMaterial({
      uniforms,
      vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix*modelViewMatrix*vec4(position,1.0); }`,
      fragmentShader: frag,
      transparent: true,
      depthWrite: false,
      depthTest: false,
      blending: blend,
    })
  );
  m.renderOrder = order;
  return m;
}

const sky = billboard(200, `${GLSL}
  varying vec2 vUv;
  void main(){
    vec2 q = (vUv-0.5)*2.0;
    // horizon-ish vertical ramp plus a wide bloom behind the disc
    float v = ssm(-0.9, 0.55, q.y);
    vec3 col = mix(vec3(0.055,0.0040,0.0022), vec3(0.255,0.0175,0.0068), v);
    col += vec3(0.430,0.0300,0.0105) * exp(-pow(length(q*vec2(1.0,1.25) - vec2(0.0,0.06))/0.34, 2.0));
    col *= 1.0 - 0.30*vnoise(q*3.1);
    col *= 1.0 - 0.50*ssm(0.30, 1.30, length(q));
    gl_FragColor = vec4(col, 1.0);
  }`, {}, -30, THREE.NormalBlending);
sky.position.set(0, CY, -60);
scene.add(sky);

const hazeMat = new THREE.ShaderMaterial({
  uniforms: { uT: { value: 0 }, uSwirl: { value: 0 }, uA: { value: 1 } },
  vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix*modelViewMatrix*vec4(position,1.0); }`,
  fragmentShader: `${GLSL}
    varying vec2 vUv; uniform float uT, uSwirl, uA;
    void main(){
      vec2 q = (vUv-0.5)*2.0;
      float d = length(q);
      if (d > 1.0) discard;
      vec2 p = rot2(q, uSwirl*0.55/(d+0.30) + uT*0.10*TAU) * 2.1;
      float n = fbm2(p + fbm2(p*1.7)*0.8);
      float fall = (1.0 - ssm(0.16, 1.0, d));
      float a = fall * (0.34 + 0.60*n) * uA;
      vec3 col = mix(C_HAZ*0.38, C_SHA*0.90, n*n) * (0.50 + 1.30/(1.0+pow(d*2.2,2.0)));
      gl_FragColor = vec4(col*a, a);
    }`,
  transparent: true,
  depthWrite: false,
  blending: THREE.NormalBlending,

const haze = new THREE.Mesh(new THREE.PlaneGeometry(9.0, 9.0), hazeMat);
haze.renderOrder = -10;
vortex.add(haze);

// ---------------------------------------------------------------- swirl discs
// three stacked polar-swirl discs: two alpha-blended bodies, one additive highlight.
// angle = atan(p) + twist/(dist+eps) - t*spin, and the sample radius creeps inward
// (uInflow) so the bands are pulled toward the centre instead of merely spinning.
function swirlDisc(radius, spin, twist, thrBase, opacity, seed, additive, lobeAmt) {
  const mat = new THREE.ShaderMaterial({
    uniforms: {
      uT: { value: 0 }, uSwirl: { value: 0 }, uErode: { value: 0 }, uA: { value: 0 },
      uSpin: { value: spin }, uTwist: { value: twist }, uThr: { value: thrBase },
      uSeed: { value: seed }, uLobe: { value: lobeAmt },
    },
    vertexShader: `varying vec2 vUv; void main(){ vUv = uv; gl_Position = projectionMatrix*modelViewMatrix*vec4(position,1.0); }`,
    fragmentShader: `${GLSL}
      varying vec2 vUv;
      uniform float uT, uSwirl, uErode, uA, uSpin, uTwist, uThr, uSeed, uLobe;
      void main(){
        vec2 q = (vUv-0.5)*2.0;
        float d = length(q);
        if (d > 1.0) discard;
        // polar swirl: differential rotation shears the noise into spiral bands
        float rot = uSwirl*uTwist/(d+0.08) + uT*uSpin*TAU;
        // radial flow: the sampled radius creeps outward with t, so features drift in
        vec2 p = rot2(q, rot) * (1.0 + uT*0.05*uSwirl) * 4.4 + uSeed;
        vec2 w = vec2(fbm3(p*1.15), fbm3(p*1.15 + 31.7)) - 0.5;
        float n = fbm3(p + w*1.15);
        // explicit log spiral so the arms READ; the fbm only wiggles them
        float ang = atan(q.y, q.x);
        float arms = 0.5 + 0.5*sin(6.0*(ang + uSwirl*uTwist*1.85*log(d+0.09)) + uT*uSpin*TAU*6.0 + (n-0.5)*5.6);
        // cauliflower lobes at the mask edge (lobe scale ≈ 0.3 units)
        float lobe = fbm3(p*2.6 + 57.0) - 0.5 + 0.50*(fbm2(p*6.4 + 91.0) - 0.5);
        float m = arms*0.62 + n*0.62 + lobe*uLobe;
        float outer = 1.0 - ssm(0.68, 1.02, d);
        float thr = uThr + uErode*0.55 + uErode*0.35*ssm(0.2,1.0,d);
        float mask = ssm(thr, thr+0.15, m) * outer;
        float lit = 1.0/(1.0 + pow(d*1.95, 2.0));
        vec3 col = mix(C_SHA*0.28, C_MID, ssm(0.02, 0.85, mask));
        col = mix(col, C_LIT, ssm(0.40, 1.00, lit)*ssm(0.20, 0.75, mask));
        col = mix(col, C_HI, ssm(0.80, 1.00, lit)*ssm(0.25, 0.80, mask));
        col *= 0.72 + 0.55*lit;
        col += C_HI * ssm(0.965, 1.00, lit) * 0.50;         // only the inner ~12% goes hot
        float a = mask * uA;
        if (a < 0.004) discard;
        gl_FragColor = vec4(col*a, a);
      }`,
    transparent: true,
    depthWrite: false,
    blending: additive ? THREE.AdditiveBlending : THREE.NormalBlending,
  });
  const m = new THREE.Mesh(new THREE.PlaneGeometry(radius * 2, radius * 2), mat);
  m.userData.opacity = opacity;
  return m;
}

const discs = [
  swirlDisc(R0, 0.35, 1.10, 0.50, 0.95, 0.0, false, 0.30),
  swirlDisc(R0 * 0.8, 0.50, 0.90, 0.56, 0.80, 11.3, false, 0.26),
  swirlDisc(R0 * 0.6, 0.70, 0.70, 0.64, 0.55, 23.9, true, 0.20),
];
const DISC_TILT = [-0.070, 0.070, -0.035];
discs.forEach((m, i) => {
  m.rotation.set(DISC_TILT[i], DISC_TILT[2 - i] * 0.6, 0);   // ±4° so no two rims coincide
  m.position.z = i * 0.05;
  m.renderOrder = i === 2 ? 7 : i;
  vortex.add(m);
});

// ---------------------------------------------------------------- cloud puffs
// a ring of camera-facing billboards ON the disc plane. They are what gives the
// oblique read: the far half sits behind the discs and is darker and smaller.
function instQuad(n) {
  const g = new THREE.InstancedBufferGeometry();
  const base = new THREE.PlaneGeometry(1, 1);
  g.index = base.index;
  g.attributes.position = base.attributes.position;
  g.attributes.uv = base.attributes.uv;
  g.instanceCount = n;
  return g;
}

const puffVert = `
  attribute vec3 aOff; attribute vec4 aP; attribute float aA;
  varying vec2 vUv; varying vec2 vLight; varying float vSeed, vRad, vLit, vA;
  uniform vec3 uCenter;
  void main(){
    vUv = uv; vSeed = aP.y; vRad = aP.z; vLit = aP.w; vA = aA;
    vec4 mv = modelViewMatrix * vec4(aOff, 1.0);
    vec4 cv = modelViewMatrix * vec4(uCenter, 1.0);
    vLight = normalize(cv.xy - mv.xy + vec2(1e-4));
    mv.xy += position.xy * aP.x;
    gl_Position = projectionMatrix * mv;
  }`;

const puffFrag = `${GLSL}
  varying vec2 vUv; varying vec2 vLight; varying float vSeed, vRad, vLit, vA;
  uniform float uA;
  void main(){
    vec2 q = (vUv-0.5)*2.0;
    float ang = atan(q.y, q.x);
    // cauliflower silhouette: the radius wobbles with angle
    float bump = 1.0 + 0.16*(vnoise(vec2(cos(ang),sin(ang))*1.9 + vSeed*7.0) - 0.5)
                     + 0.09*sin(ang*4.0 + vSeed*11.0) + 0.055*sin(ang*7.0 - vSeed*5.0);
    float r = length(q)/max(bump, 0.5);
    // soft, near-gaussian body: overlapping puffs have to blend, not stack as shards
    float a = pow(1.0 - ssm(0.06, 1.00, r), 1.30) * (0.36 + 0.42*vRad) * vA * uA;
    if (a < 0.004) discard;
    // toon-ish 2-band shading against the core light
    float nz = sqrt(max(0.0, 1.0 - r*r));
    float lam = dot(normalize(vec3(vLight, 0.62)), normalize(vec3(q, nz + 0.35)));
    vec3 shadow = mix(C_SHA*0.95, C_SHA*1.90, vRad);   // outer puffs are dark red
    vec3 body   = mix(C_SHA*2.30, C_MID*1.55, vRad);
    vec3 high   = mix(C_MID*1.90, C_LIT, vRad*0.80);
    vec3 col = mix(shadow, body, ssm(0.10, 0.30, lam));
    col = mix(col, high, ssm(0.58, 0.82, lam)*0.70);
    col *= 0.62 + 0.85*vLit;                            // fake core light falloff
    gl_FragColor = vec4(col*a, a);
  }`;

const puffMat = new THREE.ShaderMaterial({
  uniforms: { uA: { value: 1 }, uCenter: { value: new THREE.Vector3(0, 0, 0) } },
  vertexShader: puffVert,
  fragmentShader: puffFrag,
  transparent: true,
  depthWrite: false,
  blending: THREE.NormalBlending,
});

function makePuffMesh(order) {
  const g = instQuad(0);
  g.setAttribute("aOff", new THREE.InstancedBufferAttribute(new Float32Array(N_PUFF * 3), 3));
  g.setAttribute("aP", new THREE.InstancedBufferAttribute(new Float32Array(N_PUFF * 4), 4));
  g.setAttribute("aA", new THREE.InstancedBufferAttribute(new Float32Array(N_PUFF), 1));
  const m = new THREE.Mesh(g, puffMat);
  m.frustumCulled = false;
  m.renderOrder = order;
  vortex.add(m);
  return m;
}
const puffFar = makePuffMesh(2);    // behind the disc bodies
const puffNear = makePuffMesh(8);   // in front of the additive highlight

// per-puff constants, hashed once
const PUFF = [];
for (let i = 0; i < N_PUFF; i++) {
  const u = (i + 0.5) / N_PUFF;
  const r = 0.75 + 1.30 * Math.pow(h11(i * 3.11 + 0.7), 0.70);
  PUFF.push({
    r,
    ang0: u * Math.PI * 2 + (h11(i * 7.3) - 0.5) * 0.7,
    size: (0.36 + 0.40 * h11(i * 5.9 + 2.1)) * (1.20 - 0.28 * (r / 2.2)),
    seed: h11(i * 13.7) * 10,
    lift: (h11(i * 2.7 + 5.0) - 0.5) * 0.30,
    stag: h11(i * 9.1 + 3.3),
  });
}

// ---------------------------------------------------------------- flecks

fleckGeo.setAttribute("aQ", new THREE.InstancedBufferAttribute(new Float32Array(N_FLECK * 3), 3));

      vUv = uv; vA = aQ.z;
      vec4 mv = modelViewMatrix * vec4(aOff, 1.0);
      float c = cos(aQ.y), s = sin(aQ.y);
      vec2 o = vec2(position.x*c - position.y*s*0.45, position.x*s + position.y*c*0.45);

  fragmentShader: `${GLSL}

    vec2 q = (vUv-0.5)*2.0;
    float r = length(q);
    // hot-spot flicker: a slow noise field over the bright disc

// ---------------------------------------------------------------- post
const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));

// ---------------------------------------------------------------- per-frame
const _v = new THREE.Vector3();
const farBuf = { off: puffFar.geometry.attributes.aOff, p: puffFar.geometry.attributes.aP, a: puffFar.geometry.attributes.aA };
const nearBuf = { off: puffNear.geometry.attributes.aOff, p: puffNear.geometry.attributes.aP, a: puffNear.geometry.attributes.aA };
const order = [];

function apply(t) {
  const L = P.layers;
  const sw = swirlEnv(t);
  const er = erodeEnv(t);
  const fade = 1 - ss(5.15, T_OUT1, t);
  const discA = ss(T_DISC0, T_DISC1, t) * fade;

  discs.forEach((m) => {
    const u = m.material.uniforms;
    u.uT.value = t; u.uSwirl.value = sw; u.uErode.value = er;
    u.uA.value = m.userData.opacity * discA;
    m.visible = L.discs && u.uA.value > 1e-3;
  });

  hazeMat.uniforms.uT.value = t;
  hazeMat.uniforms.uSwirl.value = sw;
  hazeMat.uniforms.uA.value = 0.90 * ss(0.10, 1.10, t) * (1 - ss(T_OUT0, 6.0, t));
  haze.visible = L.haze && hazeMat.uniforms.uA.value > 1e-3;
  glowMat.uA.value = 0.85 * ss(0.05, 0.9, t) * (1 - ss(T_OUT0, 5.95, t));
  skyGlow.visible = L.haze && glowMat.uA.value > 1e-3;

  // core lights first and dims last
  coreU.uT.value = t;
  coreU.uA.value = ss(T_CORE0, T_CORE1, t) * (1 - ss(5.05, 5.95, t));
  coreHalo.visible = coreDisc.visible = L.core && coreU.uA.value > 1e-3;

  // puffs: closed-form orbit, then split far/near by view depth and sort each half
  vortex.updateMatrixWorld();
  const puffA = ss(0.25, 1.15, t) * fade;
  const shrink = 1 - 0.55 * er;
  order.length = 0;
  for (let i = 0; i < N_PUFF; i++) {
    const q = PUFF[i];
    const ang = q.ang0 - t * 0.52 * Math.pow(q.r / 2.2, -0.65) * Math.PI * 2 * 0.2;
    const x = Math.cos(ang) * q.r, y = Math.sin(ang) * q.r;
    const z = q.lift + 0.06 * Math.sin(t * 0.9 + q.seed);
    _v.set(x, y, z).applyMatrix4(vortex.matrixWorld).applyMatrix4(camera.matrixWorldInverse);
    const stag = ss(q.stag * 0.55, q.stag * 0.55 + 0.55, t);   // staggered scale-in
    order.push({ i, x, y, z, vz: _v.z, lit: 1 / (1 + Math.pow(q.r * 0.72, 2)), stag });
  }
  order.sort((a, b) => a.vz - b.vz);   // farthest (most negative z) first
  let nf = 0, nn = 0;
  const cz = _v.set(0, 0, 0).applyMatrix4(vortex.matrixWorld).applyMatrix4(camera.matrixWorldInverse).z;
  for (const o of order) {
    const q = PUFF[o.i];
    const far = o.vz < cz;
    const b = far ? farBuf : nearBuf;
    const k = far ? nf++ : nn++;
    b.off.array[k * 3] = o.x; b.off.array[k * 3 + 1] = o.y; b.off.array[k * 3 + 2] = o.z;
    b.p.array[k * 4] = q.size * (0.30 + 0.70 * o.stag) * shrink * (far ? 0.86 : 1.12);
    b.p.array[k * 4 + 1] = q.seed;
    b.p.array[k * 4 + 2] = 1 - clamp01((q.r - 0.8) / 1.4);     // 1 = inner, 0 = outer
    b.p.array[k * 4 + 3] = o.lit * (far ? 0.42 : 1.10);        // far side reads darker
    b.a.array[k] = o.stag * (far ? 0.80 : 1.0);
  }
  for (const b of [farBuf, nearBuf]) { b.off.needsUpdate = true; b.p.needsUpdate = true; b.a.needsUpdate = true; }
  puffFar.geometry.instanceCount = nf;
  puffNear.geometry.instanceCount = nn;
  puffMat.uniforms.uA.value = puffA;
  puffFar.visible = puffNear.visible = L.puffs && puffA > 1e-3;

  // flecks
  const fo = fleckGeo.attributes.aOff, fq = fleckGeo.attributes.aQ;
  const fleckA = ss(0.55, 1.30, t) * (1 - ss(T_OUT0, 5.70, t));
  for (let i = 0; i < N_FLECK; i++) {
    const f = FLECK[i];
    const a = f.ang0 - t * f.w;
    fo.array[i * 3] = Math.cos(a) * f.r;
    fo.array[i * 3 + 1] = Math.sin(a) * f.r * f.ecc;
    fo.array[i * 3 + 2] = f.bob * Math.sin(t * f.bobF + f.ang0);
    fq.array[i * 3] = f.size;
    fq.array[i * 3 + 1] = f.ang0 * 3.1 + t * f.spin;
    fq.array[i * 3 + 2] = fleckA;
  }
  fo.needsUpdate = true; fq.needsUpdate = true;
  flecks.visible = L.flecks && fleckA > 1e-3;

  bloom.enabled = L.post;
}

let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1000);
  last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom;
  bloom.threshold = P.threshold;
  apply(P.time);
  composer.render();
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
