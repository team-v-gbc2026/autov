const clamp01 = (x) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a, b, x) => {
  const t = clamp01((x - a) / (b - a));
  return t * t * (3 - 2 * t);
};
// piecewise-linear envelope over [t,v] pairs
const env = (t, pts) => {
  if (t <= pts[0][0]) return pts[0][1];
  for (let i = 1; i < pts.length; i++) {
    if (t <= pts[i][0]) {
      const [t0, v0] = pts[i - 1];
      const [t1, v1] = pts[i];
      return v0 + (v1 - v0) * ss(0, 1, (t - t0) / (t1 - t0));
    }
  }
  return pts[pts.length - 1][1];
};
const hash11 = (p) => {
  p = (p * 0.1031) % 1;
  if (p < 0) p += 1;
  p *= p + 33.33;
  p *= p + p;
  return p - Math.floor(p);
};

// head parameter: eased 0..1 across the travel window
const headU = (t) => ss(0, 1, (t - T_LAUNCH) / T_TRAVEL);
// inverse of the easing, so a trail instance knows exactly when the head passed it
const invSS = (y) => 0.5 - Math.sin(Math.asin(1 - 2 * clamp01(y)) / 3);
const bez = (u, out) => {
  const m = 1 - u;
  return out.set(
    m * m * A.x + 2 * m * u * C.x + u * u * B.x,
    m * m * A.y + 2 * m * u * C.y + u * u * B.y,
    m * m * A.z + 2 * m * u * C.z + u * u * B.z
  );
};

// ---------------------------------------------------------------- shared glsl
const GLSL_HASH = `
float hash11(float p){ p = fract(p*0.1031); p *= p+33.33; p *= p+p; return fract(p); }
float hash21(vec2 p){ vec3 q = fract(vec3(p.xyx)*0.1031); q += dot(q, q.yzx+33.33); return fract((q.x+q.y)*q.z); }
float vnoise(vec2 p){
  vec2 i = floor(p), f = fract(p);
  float a = hash21(i), b = hash21(i+vec2(1.0,0.0));
  float c = hash21(i+vec2(0.0,1.0)), d = hash21(i+vec2(1.0,1.0));
  vec2 u = f*f*(3.0-2.0*f);
  return mix(mix(a,b,u.x), mix(c,d,u.x), u.y);
}`;

const GLSL_PATH = `
const vec3 PA = vec3(${A.x}, ${A.y}, ${A.z});
const vec3 PB = vec3(${B.x}, ${B.y}, ${B.z});
const vec3 PC = vec3(${C.x}, ${C.y}, ${C.z});
vec3 bez(float u){ float m = 1.0-u; return m*m*PA + 2.0*m*u*PC + u*u*PB; }
vec3 bezT(float u){ return normalize(2.0*(1.0-u)*(PC-PA) + 2.0*u*(PB-PC)); }`;

const billboard = `
  vec3 bbRight = vec3(modelViewMatrix[0][0], modelViewMatrix[1][0], modelViewMatrix[2][0]);
  vec3 bbUp    = vec3(modelViewMatrix[0][1], modelViewMatrix[1][1], modelViewMatrix[2][1]);`;

function instQuads(n, seedFill) {
  const base = new THREE.PlaneGeometry(1, 1);
  const g = new THREE.InstancedBufferGeometry();
  g.index = base.index;
  g.attributes.position = base.attributes.position;
  g.attributes.uv = base.attributes.uv;
  const seeds = new Float32Array(n);
  for (let i = 0; i < n; i++) seeds[i] = seedFill ? seedFill(i) : i;
  g.setAttribute("aSeed", new THREE.InstancedBufferAttribute(seeds, 1));
  g.instanceCount = n;
  g.boundingSphere = new THREE.Sphere(new THREE.Vector3(0, 1.6, 0), 9);
  return g;
}

// ---------------------------------------------------------------- trail
// Path-anchored instances: each quad owns a fixed u_i and the exact time the
// head passed it. It holds that world position, flickers, then fades out.
const TRAIL = 156;
function buildTrail() {
  const g = instQuads(TRAIL);
  const us = new Float32Array(TRAIL);
  const births = new Float32Array(TRAIL);
  for (let i = 0; i < TRAIL; i++) {
    const u = (i + 0.5) / TRAIL;
    us[i] = u;
    births[i] = T_LAUNCH + T_TRAVEL * invSS(u);
  }
  g.setAttribute("aU", new THREE.InstancedBufferAttribute(us, 1));
  g.setAttribute("aBirth", new THREE.InstancedBufferAttribute(births, 1));
  return g;
}
const trailMat = new THREE.ShaderMaterial({
  uniforms: {
    uTime: { value: 0 },
    uI: { value: 1 },
    uCyan: { value: new THREE.Color("#3fe6ff") },
    uViolet: { value: new THREE.Color("#8f5cff") },
    uMagenta: { value: new THREE.Color("#ff46d2") },
  },
  vertexShader: `
    attribute float aSeed; attribute float aU; attribute float aBirth;
    uniform float uTime;
    uniform vec3 uCyan, uViolet, uMagenta;
    varying vec2 vQ; varying float vA; varying float vRect; varying vec3 vCol;
    ${GLSL_HASH}
    ${GLSL_PATH}
    void main(){
      float s = aSeed;
      float h1 = hash11(s*1.37+0.11), h2 = hash11(s*2.71+5.3), h3 = hash11(s*3.93+11.7);
      float h4 = hash11(s*5.11+23.9), h5 = hash11(s*7.13+41.3);
      float age = uTime - aBirth;
      float life = 1.00 + 0.55*h2;
      float u = age/life;
      float alive = step(0.0, age) * step(u, 1.0);

      vec3 p = bez(aU);
      vec3 tg = bezT(aU);
      vec3 side = normalize(cross(tg, vec3(0.0,1.0,0.0)));
      vec3 upn = normalize(cross(side, tg));
      // scatter off the centre line so the trail reads as a dotted band
      p += side*(h3-0.5)*0.22 + upn*(h4-0.5)*0.17;
      // ~15% are broken fragments that jump sideways in stepped windows
      float brk = step(0.85, h5);
      float st = floor(uTime*9.0);
      p += side * brk * (hash11(st*5.7 + s*3.3) - 0.5) * 0.34 * step(0.35, hash11(st*2.1 + s*9.1));

      // hold bright, then flicker out
      float fade = 1.0 - smoothstep(0.62, 1.0, u);
      float fl = step(0.30, hash11(floor(uTime*12.0)*2.7 + s*13.1));
      fl = mix(1.0, fl, smoothstep(0.45, 0.85, u));
      vA = alive * fade * fl * (0.75 + 0.25*sin(uTime*24.0 + s*7.0));
      vRect = step(0.35, h1);
      vCol = mix(mix(uCyan, uViolet, step(0.28, h2)), uMagenta, step(0.86, h2));

      ${billboard}
      // rectangles lie along the tangent; project it to screen to pick the angle
      vec3 tv = normalize((modelViewMatrix * vec4(tg, 0.0)).xyz);
      float rot = atan(tv.y, tv.x) + (h4-0.5)*0.5;
      vec2 q = position.xy*2.0;
      vec2 sc = mix(vec2(1.0,1.0), vec2(2.3,0.42), vRect);
      vec2 qs = q*sc;
      vec2 qr = vec2(qs.x*cos(rot)-qs.y*sin(rot), qs.x*sin(rot)+qs.y*cos(rot));
      vQ = q;
      float size = (0.013 + 0.030*h3) * mix(1.1, 0.65, u);
      gl_Position = projectionMatrix * modelViewMatrix * vec4(p + (bbRight*qr.x + bbUp*qr.y)*size*alive, 1.0);
    }`,
  fragmentShader: `
    varying vec2 vQ; varying float vA; varying float vRect; varying vec3 vCol;
    uniform float uI;
    void main(){
      float m;
      if (vRect > 0.5) m = step(max(abs(vQ.x), abs(vQ.y)), 0.9);
      else m = pow(max(0.0, 1.0 - length(vQ)*1.05), 2.2);
      float al = clamp(m, 0.0, 1.0) * vA * uI;
      gl_FragColor = vec4(vCol*al*1.25, al);
    }`,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
});
const trail = new THREE.Mesh(buildTrail(), trailMat);
scene.add(trail);

// ---------------------------------------------------------------- hairlines
// Thin additive strips filling the window just behind the head.
const HAIR = 3;
const HSEG = 96;
function buildHair() {
  const g = new THREE.BufferGeometry();
  const vtx = HAIR * (HSEG + 1) * 2;
  const pos = new Float32Array(vtx * 3);
  const aW = new Float32Array(vtx);
  const aSide = new Float32Array(vtx);
  const aStrand = new Float32Array(vtx);
  const idx = [];
  let v = 0;
  for (let s = 0; s < HAIR; s++) {
    const base = v;
    for (let i = 0; i <= HSEG; i++) {
      for (let k = 0; k < 2; k++) {
        aW[v] = i / HSEG;
        aSide[v] = k === 0 ? -1 : 1;
        aStrand[v] = s;
        v++;
      }
    }
    for (let i = 0; i < HSEG; i++) {
      const a = base + i * 2;
      idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
    }
  }
  g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  g.setAttribute("aW", new THREE.BufferAttribute(aW, 1));
  g.setAttribute("aSide", new THREE.BufferAttribute(aSide, 1));
  g.setAttribute("aStrand", new THREE.BufferAttribute(aStrand, 1));
  g.setIndex(idx);
  g.boundingSphere = new THREE.Sphere(new THREE.Vector3(0, 1.8, 0), 9);
  return g;
}
const hairMat = new THREE.ShaderMaterial({
  uniforms: {
    uTime: { value: 0 },
    uHead: { value: 0 },
    uTail: { value: 0.42 },
    uAlpha: { value: 0 },
    uCore: { value: new THREE.Color("#cfe8ff") },
    uEdge: { value: new THREE.Color("#8f5cff") },
  },
  vertexShader: `
    attribute float aW; attribute float aSide; attribute float aStrand;
    uniform float uTime, uHead, uTail;
    varying float vSide; varying float vW; varying float vFade; varying float vS;
    ${GLSL_HASH}
    ${GLSL_PATH}
    void main(){
      float hs = hash11(aStrand*11.7+3.1);
      float u = clamp(uHead - (1.0 - aW)*uTail, 0.0, 1.0);
      vec3 p = bez(u);
      vec3 tg = bezT(u);
      vec3 side = normalize(cross(tg, vec3(0.0,1.0,0.0)));
      vec3 upn = normalize(cross(side, tg));
      float ph = uTime*3.0 + hs*6.28 + u*22.0;
      p += side*sin(ph)*0.045*(hs+0.4) + upn*cos(ph*0.8)*0.035*(hs+0.4);
      vec3 view = normalize(cameraPosition - p);
      vec3 across = normalize(cross(tg, view));
      float tp = smoothstep(0.0, 0.22, aW) * smoothstep(1.0, 0.92, aW);
      float w = (0.0025 + 0.003*hs) * tp;
      vFade = tp; vSide = aSide; vW = aW; vS = hs;
      gl_Position = projectionMatrix * modelViewMatrix * vec4(p + across*aSide*w, 1.0);
    }`,
  fragmentShader: `
    uniform vec3 uCore, uEdge; uniform float uAlpha, uTime;
    varying float vSide; varying float vW; varying float vFade; varying float vS;
    void main(){
      float q = abs(vSide);
      float core = exp(-q*q*10.0);
      float halo = exp(-q*q*2.0);
      float flick = 0.7 + 0.3*sin(vW*60.0 + uTime*22.0 + vS*9.0);
      vec3 col = mix(uEdge, uCore, core) * (core*1.4 + halo*0.28) * flick;
      float a = vFade * uAlpha * (core*0.85 + halo*0.25);
      gl_FragColor = vec4(col*a, a);
    }`,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
  side: THREE.DoubleSide,
});
const hair = new THREE.Mesh(buildHair(), hairMat);
scene.add(hair);

// ---------------------------------------------------------------- impact wireframes
// Angular polygon outlines + long straight spokes, scaled outward with easeOut.
function buildWire() {
  const pos = [];
  const dir = [];
  const seed = [];
  const kind = [];
  const push = (a, b, d, s, k) => {
    pos.push(a.x, a.y, a.z, b.x, b.y, b.z);
    dir.push(d.x, d.y, d.z, d.x, d.y, d.z);
    seed.push(s, s);
    kind.push(k, k);
  };
  const v = new THREE.Vector3();
  // 14 polygon outlines (triangles / quads) on random planes
  for (let k = 0; k < 20; k++) {
    const n = 3 + (hash11(k * 3.7 + 1.1) > 0.5 ? 1 : 0);
    const th = hash11(k * 1.31 + 0.7) * Math.PI * 2;
    const ph = Math.acos(1 - 2 * hash11(k * 2.17 + 5.5));
    const d = new THREE.Vector3(
      Math.sin(ph) * Math.cos(th),
      Math.cos(ph) * 0.7 + 0.28,
      Math.sin(ph) * Math.sin(th)
    ).normalize();
    const e1 = new THREE.Vector3()
      .crossVectors(d, new THREE.Vector3(0, 1, 0.001))
      .normalize();
    const e2 = new THREE.Vector3().crossVectors(d, e1).normalize();
    const r = 0.30 + 0.42 * hash11(k * 5.9 + 2.2);
    const rot = hash11(k * 7.3 + 9.1) * Math.PI * 2;
    const pts = [];
    for (let i = 0; i < n; i++) {
      const a = rot + (i / n) * Math.PI * 2;
      const rr = r * (0.72 + 0.55 * hash11(k * 11.3 + i * 2.7));
      pts.push(
        new THREE.Vector3()
          .addScaledVector(e1, Math.cos(a) * rr)
          .addScaledVector(e2, Math.sin(a) * rr)
      );
    }
    for (let i = 0; i < n; i++)
      push(pts[i], pts[(i + 1) % n], d, k + 0.5, 0);
  }
  // 18 long straight spokes radiating from the hit
  for (let k = 0; k < 14; k++) {
    const th = hash11(k * 4.13 + 17.3) * Math.PI * 2;
    const ph = Math.acos(1 - 2 * hash11(k * 6.71 + 3.9));
    const d = new THREE.Vector3(
      Math.sin(ph) * Math.cos(th),
      Math.cos(ph) * 0.85 + 0.12,
      Math.sin(ph) * Math.sin(th)
    ).normalize();
    const l0 = 0.25 + 0.5 * hash11(k * 2.9 + 8.1);
    const l1 = l0 + 0.8 + 1.4 * hash11(k * 9.7 + 4.4);
    push(v.copy(d).multiplyScalar(l0), d.clone().multiplyScalar(l1), d, k + 30.5, 1);
  }
  const g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute("aDir", new THREE.Float32BufferAttribute(dir, 3));
  g.setAttribute("aSeed", new THREE.Float32BufferAttribute(seed, 1));
  g.setAttribute("aKind", new THREE.Float32BufferAttribute(kind, 1));
  g.boundingSphere = new THREE.Sphere(B.clone(), 12);
  return g;
}

// ---------------------------------------------------------------- timeline
const _p = new THREE.Vector3();
const _p2 = new THREE.Vector3();
const _tan = new THREE.Vector3();
const _look = new THREE.Vector3();

function apply(t) {
  const L = P.layers;

  // ---- charge 0 - 0.5
  const chI = env(t, [[0, 0], [0.08, 1], [0.42, 1], [0.52, 0]]) * (L.charge ? 1 : 0);
  chargeMat.uniforms.uTime.value = t;
  chargeMat.uniforms.uI.value = chI;
  charge.visible = chI > 0.001;

  // ---- head 0.5 - 1.8
  const u = headU(t);
  const flying = t >= T_LAUNCH - 0.02 && t <= T_HIT + 0.01;
  bez(u, _p);
  bez(Math.min(1, u + 0.004), _p2);
  _tan.subVectors(_p2, _p).normalize();
  // stepped-hash jitter: fires only in discrete windows, and only while flying
  const stw = Math.floor(t * 10);
  if (flying && hash11(stw * 1.7 + 0.3) > 0.85) {
    const j = 0.09;
    _p.x += (hash11(stw * 3.1) - 0.5) * j;
    _p.y += (hash11(stw * 5.3) - 0.5) * j;
    _p.z += (hash11(stw * 7.9) - 0.5) * j;
  }
  const headI = env(t, [[0.42, 0], [0.52, 1], [1.72, 1], [1.8, 0]]) * (L.head ? 1 : 0);
  dart.position.copy(_p);
  dart.lookAt(_look.copy(_p).add(_tan));
  dart.scale.set(1, 1, 1 + 0.5 * Math.sin(t * 30) * 0.1);
  dart.visible = headI > 0.01;
  dartMat.opacity = headI;
  core.position.copy(_p);
  coreMat.uniforms.uI.value = headI * 0.30;
  core.visible = headI > 0.01;
  tailMat.uniforms.uTime.value = t;
  tailMat.uniforms.uI.value = headI;
  tailMat.uniforms.uPos.value.copy(_p);
  tailMat.uniforms.uTan.value.copy(_tan);
  tails.visible = headI > 0.01 && L.head;

  // ---- trail (path-anchored, holds in space, then flickers out)
  trailMat.uniforms.uTime.value = t;
  trailMat.uniforms.uI.value = L.trail ? 1 : 0;
  trail.visible = L.trail && t > T_LAUNCH && t < 2.9;
  hairMat.uniforms.uTime.value = t;
  hairMat.uniforms.uHead.value = u;
  hairMat.uniforms.uTail.value = 0.30 + 0.16 * ss(0.5, 1.4, t);
  const hairA = env(t, [[0.48, 0], [0.6, 0.32], [1.7, 0.32], [1.85, 0]]) * (L.trail ? 1 : 0);
  hairMat.uniforms.uAlpha.value = hairA;
  hair.visible = hairA > 0.001;

  // ---- impact 1.8 - 2.1
  const e = (t - T_HIT) / 0.85; // wireframe life runs a little past the flash
  const flashI = env(t, [[1.78, 0], [1.82, 0.55], [1.9, 0.38], [2.02, 0.18], [2.3, 0]]);
  flashMat.uniforms.uTime.value = t;
  flashMat.uniforms.uI.value = flashI;
  flashMat.uniforms.uGrid.value = env(t, [[1.82, 0], [1.92, 1], [2.3, 1], [2.7, 0]]);
  flash.visible = flashI > 0.001;
  flashMat.uniforms.uScale.value = env(t, [[1.78, 0.7], [1.95, 1.5], [2.4, 1.95]]);

  const wireI = env(t, [[1.79, 0], [1.83, 1], [2.5, 0.55], [3.5, 0.12], [3.9, 0]]) * (L.impactLines ? 1 : 0);
  wireMat.uniforms.uTime.value = t;
  wireMat.uniforms.uE.value = clamp01(e);
  wireMat.uniforms.uI.value = wireI;
  wire.visible = wireI > 0.001;
  const domeI = env(t, [[1.8, 0], [1.88, 0.5], [2.2, 0.24], [2.7, 0]]) * (L.impactLines ? 1 : 0);
  domeMat.uniforms.uI.value = domeI;
  domeMat.uniforms.uTime.value = t;
  dome.visible = domeI > 0.001;
  dome.scale.setScalar(0.7 + 1.55 * (1 - Math.pow(1 - clamp01((t - 1.8) / 0.7), 2.4)));

  const shardI = (L.shards ? 1 : 0) * env(t, [[1.79, 0], [1.82, 1], [2.6, 1], [3.0, 0]]);
  shardMat.uniforms.uTime.value = t;
  shardMat.uniforms.uE.value = Math.max(0, t - T_HIT);
  shardMat.uniforms.uI.value = shardI;
  shards.visible = shardI > 0.001 && t > T_HIT;

  // ---- screen glitch: two hot frames at the hit, then a short tail
  const gI = env(t, [[1.79, 0], [1.81, 1], [1.9, 0.9], [1.95, 0.35], [2.0, 0]]) * (L.screenGlitch ? 1 : 0);
  glitchPass.uniforms.uTime.value = t;
  glitchPass.uniforms.uI.value = gI;
  glitchPass.enabled = gI > 0.002;

  // ---- aftermath 2.1 - 4.0
  const smokeI = env(t, [[1.92, 0], [2.25, 0.7], [2.9, 0.62], [3.55, 0.25], [3.95, 0]]) * (L.smoke ? 1 : 0);
  const sr = env(t, [[1.9, 0.34], [2.5, 0.80], [3.9, 1.05]]);
  smokeMat.uniforms.uTime.value = t;
  smokeMat.uniforms.uI.value = smokeI;
  smokeMat.uniforms.uR.value = sr * 4.5;
  smokeMat.uniforms.uAsp.value = 1.0;
  smoke.visible = smokeI > 0.001;
  puffMat.uniforms.uTime.value = t;
  puffMat.uniforms.uE.value = clamp01((t - 1.9) / 1.5);
  puffMat.uniforms.uI.value = env(t, [[1.9, 0], [2.2, 0.60], [3.0, 0.40], [3.6, 0.14], [3.95, 0]]) * (L.smoke ? 1 : 0);
  puffs.visible = puffMat.uniforms.uI.value > 0.001;

  const gboxI = env(t, [[1.86, 0], [1.95, 0.9], [2.9, 0.7], [3.7, 0.2], [3.95, 0]]) * (L.impactLines ? 1 : 0);
  gboxMat.uniforms.uTime.value = t;
  gboxMat.uniforms.uI.value = gboxI;
  gbox.visible = gboxI > 0.001;

  const sparkI = env(t, [[2.0, 0], [2.3, 1], [3.4, 0.8], [3.9, 0]]) * (L.smoke ? 1 : 0);
  sparkMat.uniforms.uTime.value = t;
  sparkMat.uniforms.uI.value = sparkI;
  sparks.visible = sparkI > 0.001;

  // ground bounce from the hit
  ground.material.uniforms.uTintI.value = flashI * 0.55 + smokeI * 0.05;
}

let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1000);
  last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom;
  bloom.threshold = P.threshold;
  apply(P.time);
  composer.render();
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

window.spikeSetTime = (t) => {
  P.play = false;
  P.time = t;
  apply(t);
};
window.__spikeReady = true;
