const QUAD = new THREE.PlaneGeometry(1, 1);

function addCard(frag, uniforms, order, blend) {
  const m = new THREE.ShaderMaterial({
    transparent: true, depthWrite: false, depthTest: false,
    blending: blend === "alpha" ? THREE.NormalBlending : THREE.AdditiveBlending,
    uniforms: Object.assign({ uA: { value: 1 } }, uniforms),
    vertexShader: /* glsl */`varying vec2 vUv; void main(){ vUv=uv*2.-1.; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.); }`,
    fragmentShader: frag,
  });
  const mesh = new THREE.Mesh(QUAD, m);
  mesh.renderOrder = order;
  burst.add(mesh);
  return mesh;
}

// ---------------------------------------------------------------- flash
// The reference's centre is a DRAWN shape, not a blob: a big solid 5-point star,
// saturated pink, thick white outline, screentone inside, a white-hot star within it.
// It is alpha-blended (additive would bleach the pink straight back to white), and the
// outline outlives the hot core — the shape is still legible long after the flash.
const GLSL_STAR5 = /* glsl */`
  float sdStar5(vec2 p, float r, float rf){
    const vec2 k1=vec2(0.809016994375,-0.587785252292);
    const vec2 k2=vec2(-k1.x,k1.y);
    p.x=abs(p.x);
    p-=2.0*max(dot(k1,p),0.0)*k1;
    p-=2.0*max(dot(k2,p),0.0)*k2;
    p.x=abs(p.x); p.y-=r;
    vec2 ba=rf*vec2(-k1.y,k1.x)-vec2(0,1);
    float h=clamp(dot(p,ba)/dot(ba,ba),0.0,r);
    return length(p-ba*h)*sign(p.y*ba.x-p.x*ba.y);
  }`;

const solidStar = addCard(/* glsl */`
  precision highp float; varying vec2 vUv;
  uniform float uA,uFillA,uHotA,uRot,uOut,uTone;
  uniform vec3 uFill,uDot,uLine,uHot;
  ${GLSL_STAR5}
  void main(){
    float c=cos(uRot), si=sin(uRot);
    vec2 p=mat2(c,-si,si,c)*vUv;
    float aa=fwidth(vUv.x)*1.2+0.004;
    float d=sdStar5(p,0.70,0.46);
    float fill=smoothstep(aa,-aa,d);
    float ring=smoothstep(aa,-aa,d-uOut)-fill;        // thick white outline, outside the fill
    if(fill+ring<0.004) discard;
    // screentone: a magenta dot lattice inside the fill only
    vec2 g=fract(vUv/uTone)-0.5;
    float tone=smoothstep(0.36,0.24,length(g))*smoothstep(0.0,0.35,-d);
    vec3 col=mix(uFill,uDot,tone*0.55);
    float hot=smoothstep(aa,-aa,sdStar5(p,0.34,0.46));
    col=mix(col,uHot,hot*uHotA);
    col=mix(col,uLine,ring);
    float a=(fill*uFillA+ring)*uA;
    gl_FragColor=vec4(col*(1.0+hot*uHotA*0.85),clamp(a,0.,1.));
  }`, {
    uFillA: { value: 1 }, uHotA: { value: 1 }, uRot: { value: -0.12 }, uOut: { value: 0.105 }, uTone: { value: 0.075 },
    uFill: { value: hex("#ff4fc4") }, uDot: { value: hex("#c81f8e") },
    uLine: { value: hex("#fff2fb") }, uHot: { value: hex("#fff0cf") },
  }, 38, "alpha");

// ---------------------------------------------------------------- star lines
// flat unlit tapered slivers, same construction as the smoke spike's splash shards:
// pointed at the tip, widest a third of the way up, independently jagged edges.
function sliverGeo(len, width, curve, taper, h) {
  const N = 28, pos = [], uv = [], idx = [];
  const nL = 2 + Math.floor(hash(h) * 3), nR = 2 + Math.floor(hash(h + 7) * 3);
  const oL = hash(h + 3), oR = hash(h + 11);
  const dL = 0.16 + 0.22 * hash(h + 5), dR = 0.16 + 0.22 * hash(h + 13);
  const edge = (t, base, n, off, depth) => base * (1 - depth * ((t * n + off) % 1));
  for (let i = 0; i <= N; i++) {
    const t = i / N;
    const x = (Math.sin(Math.PI * t * 0.5) - 0.25 * Math.sin(Math.PI * t)) * curve;
    const y = len * t;
    const base = width * Math.pow(1 - t, taper) * Math.pow(ss(0, 0.22, t), 0.6);
    pos.push(x - edge(t, base, nL, oL, dL), y, 0, x + edge(t, base, nR, oR, dR), y, 0);
    uv.push(0, t, 1, t);
  }
  for (let i = 0; i < N; i++) { const k = i * 2; idx.push(k, k + 1, k + 2, k + 1, k + 3, k + 2); }
  const g = new THREE.BufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
  g.setIndex(idx);
  return g;
}

const sliverFrag = /* glsl */`
  precision highp float; varying vec2 vUv; uniform vec3 uRoot,uTip; uniform float uA,uCut;
  void main(){
    if (vUv.y < uCut) discard;                        // the shrink eats the sliver from its root
    vec3 c=mix(uRoot,uTip,smoothstep(0.05,0.75,vUv.y));
    float a=uA*(1.0-smoothstep(0.72,1.0,vUv.y)*0.55);
    a*=smoothstep(0.0,0.30,vUv.x)*smoothstep(1.0,0.70,vUv.x);   // soft edges along the width
    gl_FragColor=vec4(c,a);
  }`;

const lines = [];
for (let k = 0; k < N_LINE + N_FAT; k++) {
  const fat = k >= N_LINE;
  const h = k * 13 + 3;
  const i = fat ? k - N_LINE : k;
  const n = fat ? N_FAT : N_LINE;
  // evenly spread with a hashed wobble so the fan never reads as a clock face
  const ang = (i / n) * Math.PI * 2 + (fat ? 0.42 : 0) + (hash(h) - 0.5) * (Math.PI * 1.1 / n);
  const len = fat ? 0.40 + 0.28 * hash(h + 1) : 1.20 + 1.00 * hash(h + 1);
  const wid = fat ? 0.105 + 0.05 * hash(h + 2) : 0.055 + 0.055 * hash(h + 2);
  const curve = (hash(h + 4) < 0.45 ? 1 : -1) * (0.02 + 0.16 * hash(h + 5));  // some curved, most straight
  const g = sliverGeo(len, wid, curve, fat ? 1.15 : 2.05, h);   // high taper = fat root, needle tip
  const m = new THREE.ShaderMaterial({
    transparent: true, depthWrite: false, depthTest: false, side: THREE.DoubleSide,
    blending: THREE.AdditiveBlending,
    uniforms: {
      uRoot: { value: fat ? hex("#ff7ad2") : hex("#ffe8fb") },
      uTip: { value: fat ? hex("#ff2fa8") : hex("#b083ff") },
      uA: { value: 1 }, uCut: { value: 0 },
    },
    vertexShader: /* glsl */`varying vec2 vUv; void main(){ vUv=uv; gl_Position=projectionMatrix*modelViewMatrix*vec4(position,1.); }`,
    fragmentShader: sliverFrag,
  });
  const mesh = new THREE.Mesh(g, m);
  mesh.renderOrder = fat ? 30 : 28;
  burst.add(mesh);
  lines.push({
    mesh, m, fat, ang, len, h,
    t0: 0.245 + 0.045 * hash(h + 6),
    grow: 0.075 + 0.05 * hash(h + 7),
    r0: (fat ? 0.20 : 0.16) + 0.16 * hash(h + 8),      // inner gap so the core stays readable
    push: 0.30 + 0.45 * hash(h + 9),                    // how far the root travels out
    s0: 0.33 + 0.10 * hash(h + 10),                     // root-shrink start
    s1: 0.50 + 0.16 * hash(h + 11),
    spin: (hash(h + 12) - 0.5) * 0.6,
  });
}

// ---------------------------------------------------------------- faces
// 6 instanced camera-facing quads; the whole face is drawn in the fragment shader,
// so there is no texture and expressions are a hashed branch, not an atlas tile.
const faceMat = new THREE.ShaderMaterial({
  transparent: true, depthWrite: false, depthTest: false, side: THREE.DoubleSide,
  uniforms: {
    uFill: { value: hex("#ff3cb4") }, uHi: { value: hex("#ff8ad8") },
    uLine: { value: hex("#fff0fa") }, uEye: { value: hex("#4a0b33") },
  },
  vertexShader: /* glsl */`
    attribute float aSeed; attribute float aAlpha;
    varying vec2 vUv; varying float vSeed; varying float vAlpha;
    void main(){ vUv=uv*2.-1.; vSeed=aSeed; vAlpha=aAlpha;
      gl_Position=projectionMatrix*modelViewMatrix*instanceMatrix*vec4(position,1.); }`,
  fragmentShader: /* glsl */`
    precision highp float;
    varying vec2 vUv; varying float vSeed; varying float vAlpha;
    uniform vec3 uFill,uHi,uLine,uEye;
    float sdC(vec2 p,float r){ return length(p)-r; }
    float sdE(vec2 p,vec2 r){ return (length(p/r)-1.0)*min(r.x,r.y); }
    void main(){
      vec2 p=vUv;
      float aa=fwidth(p.x)*1.4+0.004;
      // head + two ears
      float d=sdC(p,0.58);
      d=min(d,sdC(p-vec2(-0.44,0.46),0.235));
      d=min(d,sdC(p-vec2( 0.44,0.46),0.235));
      float body=smoothstep(aa,-aa,d);
      float ring=smoothstep(aa,-aa,d-0.085)-body;      // white outline hugging the silhouette
      if(body+ring<0.003) discard;

      float expr=floor(fract(vSeed*7.31)*3.0);          // smile / ^^ / open mouth
      // eyes
      vec2 el=p-vec2(-0.235,0.115), er=p-vec2(0.235,0.115);
      float eye;
      if(expr<1.5 && expr>0.5){                          // ^^ happy eyes: two arcs
        eye=min(abs(sdC(el,0.135))-0.042, abs(sdC(er,0.135))-0.042);
        eye=max(eye,-(p.y-0.085));
      } else {
        eye=min(sdE(el,vec2(0.088,0.115)), sdE(er,vec2(0.088,0.115)));
      }
      // mouth
      float mouth;
      if(expr>1.5){ mouth=sdE(p-vec2(0.0,-0.185),vec2(0.115,0.095)); }
      else { mouth=max(abs(sdC(p-vec2(0.0,0.12),0.30))-0.038, -(-p.y-0.055)); }
      float ink=smoothstep(aa,-aa,min(eye,mouth));

      // a soft lit lift on the upper-left so the head reads as a ball, not a disc
      float lift=smoothstep(0.55,-0.35,length(p-vec2(-0.16,0.20)));
      vec3 c=mix(uFill,uHi,lift*0.55);
      // ear inners and a muzzle: the same lighter pink, so the head reads as a bear
      float inner=min(sdC(p-vec2(-0.44,0.47),0.115), sdC(p-vec2(0.44,0.47),0.115));
      float muzzle=sdE(p-vec2(0.0,-0.115),vec2(0.235,0.175));
      c=mix(c,uHi,smoothstep(aa,-aa,min(inner,muzzle))*0.85);
      c=mix(c,uLine,ring);
      c=mix(c,uEye,ink*body);
      float a=(body+ring)*vAlpha;
      gl_FragColor=vec4(c,clamp(a,0.,1.));
    }`,
});

    uniform vec3 uFill,uLine;
    float sdC(vec2 p,float r){ return length(p)-r; }
    float d2(vec2 v){ return dot(v,v); }
    // iq's heart SDF: cusp at the origin, lobes at y=1
    float sdHeart(vec2 p){
      p.x=abs(p.x);
      if(p.y+p.x>1.0) return sqrt(d2(p-vec2(0.25,0.75)))-sqrt(2.0)/4.0;
      return sqrt(min(d2(p-vec2(0.0,1.0)), d2(p-0.5*max(p.x+p.y,0.0))))*sign(p.x-p.y);
    }
    void main(){
      vec2 p=vUv; float aa=fwidth(p.x)*1.5+0.006; float d;
      if(fract(vSeed*3.7)<0.62){
        d=sdHeart((p+vec2(0.0,0.78))/1.55)*1.55;
      } else {
        // comma: a crescent, a big disc minus an offset one
        d=max(sdC(p-vec2(-0.06,0.02),0.56), -sdC(p-vec2(0.26,0.24),0.50));
      }
      float body=smoothstep(aa,-aa,d);
      float ring=smoothstep(aa,-aa,d-0.10)-body;
      if(body+ring<0.004) discard;
      vec3 c=mix(uFill,uLine,ring*0.9);
      gl_FragColor=vec4(c,clamp((body+ring)*vAlpha,0.,1.));
    }`,
});
const heartGeo = new THREE.PlaneGeometry(1, 1);
heartGeo.setAttribute("aSeed", new THREE.InstancedBufferAttribute(new Float32Array(N_HEART), 1));
heartGeo.setAttribute("aAlpha", new THREE.InstancedBufferAttribute(new Float32Array(N_HEART), 1));

hearts.renderOrder = 46;

  vertexShader: /* glsl */`
    attribute float aAlpha; varying vec2 vUv; varying float vAlpha;

    float sdSeg(vec2 p,vec2 a,vec2 b){ vec2 pa=p-a,ba=b-a; float h=clamp(dot(pa,ba)/dot(ba,ba),0.,1.); return length(pa-ba*h); }
    void main(){

// ---------------------------------------------------------------- apply(t)
const _m = new THREE.Matrix4(), _p = new THREE.Vector3(), _q = new THREE.Quaternion(), _s = new THREE.Vector3();
const _e = new THREE.Euler();

function setInst(mesh, i, x, y, rot, sx, sy) {
  _p.set(x, y, 0);
  _e.set(0, 0, rot); _q.setFromEuler(_e);
  _s.set(sx, sy, 1);
  _m.compose(_p, _q, _s);
  mesh.setMatrixAt(i, _m);
}

function apply(t) {
  const L = P.layers;
  burst.quaternion.copy(camera.quaternion);

  // ---- flash: easeOutExpo up, then a hard cut; the core is the only thing over 1.0
  const up = easeOutExpo(clamp01((t - T_ANT) / 0.085));
  const cut = 1 - ss(0.30, 0.44, t);
  const fA = up * cut;
  const fs = (0.40 + 1.25 * up) * (1 + 0.34 * (1 - cut));
  starCard.scale.set(fs, fs, 1);
  starCard.material.uniforms.uA.value = L.flash ? fA * 0.92 : 0;
  starCard.material.uniforms.uPoint.value = 0.72 + 0.28 * up;
  starCard.visible = L.flash && fA > 0.004;

  // the solid star: ~1.6 units across at peak, the hot inner star cuts with the flash,
  // the outlined shell keeps reading until 0.7 — that is where the mass comes from.
  const stA = ss(T_ANT, T_ANT + 0.07, t);
  const stFade = 1 - ss(0.44, 0.70, t);
  const stW = 1.60 * (0.42 + 0.58 * easeOutExpo(clamp01((t - T_ANT) / 0.10))) * (1 + 0.16 * ss(0.30, 0.62, t));
  solidStar.scale.set(stW, stW, 1);
  solidStar.material.uniforms.uA.value = L.flash ? stA * stFade : 0;
  solidStar.material.uniforms.uFillA.value = 0.88 * (1 - 0.55 * ss(0.40, 0.68, t));
  solidStar.material.uniforms.uHotA.value = 1 - ss(0.30, 0.44, t);
  solidStar.material.uniforms.uTone.value = 0.06 / (stW * 0.5);   // screentone pitch is WORLD-fixed
  solidStar.material.uniforms.uRot.value = -0.12 + 0.10 * ss(T_ANT, 0.7, t);
  solidStar.material.uniforms.uOut.value = 0.105 * (1 - 0.55 * ss(0.38, 0.70, t));
  solidStar.visible = L.flash && stA * stFade > 0.006;

  const hA = ss(0.26, 0.34, t) * (1 - ss(0.62, 1.00, t));
  const hs = 2.20 * (0.72 + 0.28 * ss(0.26, 0.60, t));
  haloCard.scale.set(hs, hs, 1);
  haloCard.material.uniforms.uA.value = L.flash ? hA * 0.92 : 0;
  haloCard.visible = L.flash && hA > 0.004;

  const rA = ss(0.22, 0.34, t) * (1 - ss(0.80, 1.80, t));
  const rs = 2.6 + 2.2 * ss(0.25, 1.4, t);
  residualCard.scale.set(rs, rs, 1);
  residualCard.material.uniforms.uA.value = L.flash ? rA * 0.42 : 0;
  residualCard.visible = L.flash && rA > 0.004;

  // ---- anticipation dot + inward sparkles
  const dA = ss(0.0, 0.06, t) * (1 - ss(0.19, 0.27, t));
  const ds = 0.14 + 0.30 * ss(0, T_ANT, t);
  dotCard.scale.set(ds, ds, 1);
  dotCard.material.uniforms.uA.value = L.flash ? dA * 0.9 : 0;
  dotCard.visible = L.flash && dA > 0.004;

  {
    const arr = antGeo.attributes.aAlpha.array;
    const pull = ss(0.02, T_ANT, t);
    const aA = ss(0.01, 0.07, t) * (1 - ss(0.16, 0.21, t));
    for (let i = 0; i < 8; i++) {
      const h = i * 41 + 3;
      const ang = (i / 8) * Math.PI * 2 + hash(h) * 0.5;
      const r = (0.62 + 0.28 * hash(h + 1)) * (1 - pull) + 0.06;
      const sz = 0.075 * (1 - 0.45 * pull);
      setInst(antSparks, i, Math.cos(ang) * r, Math.sin(ang) * r, 0, sz, sz);
      arr[i] = L.flash ? aA : 0;
    }
    antSparks.instanceMatrix.needsUpdate = true;
    antGeo.attributes.aAlpha.needsUpdate = true;
    antSparks.visible = L.flash && aA > 0.004;
  }

  // ---- star lines: scale out from the centre, then shrink from the root
  for (const l of lines) {
    const age = t - l.t0;
    const g = easeOutCubic(age / l.grow);
    const eat = ss(l.s0, l.s1, t);                       // 0..1 of the sliver eaten from the root
    const fade = 1 - ss(l.s1 - 0.10, l.s1 + 0.14, t);
    const vis = L.lines && age > 0 && fade > 0.01 && eat < 0.995;
    l.mesh.visible = vis;
    if (!vis) continue;
    const rr = l.r0 + l.push * easeOutCubic(clamp01(age / 0.42)) + (l.len * eat) * 0.9;
    const rot = l.ang - Math.PI / 2 + l.spin * clamp01(age) * 0.35;
    l.mesh.position.set(Math.cos(l.ang) * rr, Math.sin(l.ang) * rr, 0);
    l.mesh.rotation.set(0, 0, rot);
    l.mesh.scale.set(g * (1 - 0.35 * eat), g * (1 - eat), 1);
    // the rays duck under the flash while the core is hot, then own the frame
    const duck = 0.40 + 0.60 * ss(0.30, 0.40, t);
    l.m.uniforms.uA.value = fade * duck * (l.fat ? 0.52 : 0.38);
    l.m.uniforms.uCut.value = 0;
  }

  // ---- faces: ballistic arcs, overshoot scale-in, they fade last
  {
    const aArr = faceGeo.attributes.aAlpha.array;
    let any = false;
    for (let i = 0; i < N_FACE; i++) {
      const f = FACE[i];
      const age = t - f.t0;
      const alive = age > 0;
      const fade = 1 - ss(f.fade[0], f.fade[1], t);
      const sIn = easeOutBack(clamp01(age / 0.15));
      const a = alive ? fade : 0;
      if (a > 0.01) any = true;
      // drag-limited radial throw + a small lateral arc and a light gravity sag
      const dr = alive ? (1 - Math.exp(-age * 3.4)) / 3.4 : 0;
      const r = f.r0 + f.v0 * dr;
      const lat = f.arc * dr * 0.9;
      const cx = Math.cos(f.ang), cy = Math.sin(f.ang);
      const x = cx * r - cy * lat;
      const y = cy * r + cx * lat - 0.55 * Math.max(0, age) * Math.max(0, age);
      const wob = f.linger ? 0.16 * Math.sin(Math.max(0, age) * 7.4 + f.h) : 0;
      const sz = f.size * sIn * (1 + (f.linger ? 0.05 * Math.sin(age * 9.1) : 0));
      setInst(faces, i, x, y + wob * 0.10, f.spin0 + f.spin * Math.max(0, age) * 0.35 + wob, sz, sz);
      aArr[i] = L.faces ? a : 0;
    }
    faces.instanceMatrix.needsUpdate = true;
    faceGeo.attributes.aAlpha.needsUpdate = true;
    faces.visible = L.faces && any;
  }

  // ---- hearts / commas
  {
    const aArr = heartGeo.attributes.aAlpha.array;
    let any = false;
    for (let i = 0; i < N_HEART; i++) {
      const c = HEART[i];
      const age = t - c.t0;
      const alive = age > 0;
      const fade = 1 - ss(c.fade[0], c.fade[1], t);
      const sIn = easeOutBack(clamp01(age / 0.13));
      const a = alive ? fade : 0;
      if (a > 0.01) any = true;
      const dr = alive ? (1 - Math.exp(-age * 2.8)) / 2.8 : 0;
      const r = c.r0 + c.v0 * dr;
      const lat = c.arc * dr * 0.8;
      const cx = Math.cos(c.ang), cy = Math.sin(c.ang);
      const x = cx * r - cy * lat;
      const y = cy * r + cx * lat - 0.40 * Math.max(0, age) * Math.max(0, age);
      const sz = c.size * sIn;
      setInst(hearts, i, x, y, c.spin * Math.max(0, age) * 0.5, sz, sz);
      aArr[i] = L.commas ? a : 0;
    }
    hearts.instanceMatrix.needsUpdate = true;
    heartGeo.attributes.aAlpha.needsUpdate = true;
    hearts.visible = L.commas && any;
  }

  // ---- cloud lobes: puff outward behind the star, gone by 0.8
  {
    const aArr = lobeGeo.attributes.aAlpha.array;
    let any = false;
    for (let i = 0; i < N_LOBE; i++) {
      const b = LOBE[i];
      const age = t - b.t0;
      const fade = 1 - ss(b.fade[0], b.fade[1], t);
      const a = age > 0 ? fade : 0;
      if (a > 0.01) any = true;
      const r = b.r0 + b.push * easeOutCubic(clamp01(age / 0.34));
      const sz = b.size * easeOutBack(clamp01(age / 0.16)) * (1 - 0.25 * ss(b.fade[0], b.fade[1], t));
      setInst(lobes, i, Math.cos(b.ang) * r, Math.sin(b.ang) * r, b.spin * Math.max(0, age), sz, sz * 0.86);
      aArr[i] = L.flash ? a * 0.52 : 0;
    }
    lobes.instanceMatrix.needsUpdate = true;
    lobeGeo.attributes.aAlpha.needsUpdate = true;
    lobes.visible = L.flash && any;
  }

  // ---- lightning glyphs at the rim
  {
    const aArr = boltGeo.attributes.aAlpha.array;
    let any = false;
    for (let i = 0; i < N_BOLT; i++) {
      const b = BOLT[i];
      const age = t - b.t0;
      const fade = 1 - ss(b.fade[0], b.fade[1], t);
      const a = age > 0 ? fade * ss(0, 0.05, age) : 0;
      if (a > 0.01) any = true;
      const r = b.r0 + b.push * easeOutCubic(clamp01(age / 0.30));
      setInst(bolts, i, Math.cos(b.ang) * r, Math.sin(b.ang) * r,
        b.ang - Math.PI / 2 + b.tilt, b.size, b.size * 1.7);
      aArr[i] = L.spray ? a : 0;
    }
    bolts.instanceMatrix.needsUpdate = true;
    boltGeo.attributes.aAlpha.needsUpdate = true;
    bolts.visible = L.spray && any;
  }

  // ---- spray
  {
    const aArr = sprayGeo.attributes.aAlpha.array;
    let any = false;
    for (let i = 0; i < N_SPRAY; i++) {
      const s = SPRAY[i];
      const age = t - s.t0;
      const u = age / s.life;
      const alive = age > 0 && u < 1;
      const a = alive ? (1 - u) * (1 - Math.exp(-age * 26)) : 0;
      if (a > 0.01) any = true;
      const dr = alive ? (1 - Math.exp(-age * 3.6)) / 3.6 : 0;
      const r = s.r0 + s.v0 * dr;
      setInst(spray, i, Math.cos(s.ang) * r, Math.sin(s.ang) * r,
        s.ang, s.len * (0.6 + 0.7 * (1 - u)), s.wid);
      aArr[i] = L.spray ? a * 1.25 : 0;
    }
    spray.instanceMatrix.needsUpdate = true;
    sprayGeo.attributes.aAlpha.needsUpdate = true;
    spray.visible = L.spray && any;
  }
}

// ---------------------------------------------------------------- frame
let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1000); last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom; bloom.threshold = P.threshold;
  apply(P.time);
  if (P.layers.post) composer.render(); else renderer.render(scene, camera);
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

window.spikePieces = { lines, faces: FACE, hearts: HEART, lobes: LOBE, bolts: BOLT, spray: SPRAY, center: CENTER };
window.__spikeReady = true;
