const clamp01 = (x) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a, b, x) => {
  const t = clamp01((x - a) / (b - a));
  return t * t * (3 - 2 * t);
};
// piecewise-linear envelope over [t,v] pairs
const env = (t, pts) => {
  if (t <= pts[0][0]) return pts[0][1];
  for (let i = 1; i < pts.length; i++) {
    if (t <= pts[i][0]) {
      const [t0, v0] = pts[i - 1];
      const [t1, v1] = pts[i];
      return v0 + (v1 - v0) * ss(0, 1, (t - t0) / (t1 - t0));
    }
  }
  return pts[pts.length - 1][1];
};
// deterministic hash: the only source of per-instance variation
const h11 = (n) => {
  let p = (n * 0.1031) % 1;
  p *= p + 33.33;
  p *= p + p;
  return p - Math.floor(p);
};

// ---------------------------------------------------------------- hex lattice
// The cells are the Voronoi regions of a relaxed point set on the sphere:
// uniform, pole-free, seam-free — a Goldberg sphere without any authored mesh.
// A raw Fibonacci set tiles into rhombi, so Lloyd relaxation (a centroidal
// Voronoi tessellation) pulls it into near-regular hexagons.
const fib = (i, n) => {
  const y = 1 - (2 * i + 1) / n;
  const rr = Math.sqrt(Math.max(0, 1 - y * y));
  const ph = 2 * Math.PI * (i * 0.61803398875);
  return [Math.cos(ph) * rr, y, Math.sin(ph) * rr];
};
const site = [];
for (let i = 0; i < NPTS; i++) site.push(fib(i, NPTS));
const M = 4200;
for (let it = 0; it < 14; it++) {
  const acc = new Float64Array(NPTS * 3);
  for (let k = 0; k < M; k++) {
    const q = fib(k, M);
    let bi = 0,
      bd = 9;
    for (let i = 0; i < NPTS; i++) {
      const s2 = site[i];
      const dx = s2[0] - q[0],
        dy = s2[1] - q[1],
        dz = s2[2] - q[2];
      const d = dx * dx + dy * dy + dz * dz;
      if (d < bd) {
        bd = d;
        bi = i;
      }
    }
    acc[bi * 3] += q[0];
    acc[bi * 3 + 1] += q[1];
    acc[bi * 3 + 2] += q[2];
  }
  for (let i = 0; i < NPTS; i++) {
    const L = Math.hypot(acc[i * 3], acc[i * 3 + 1], acc[i * 3 + 2]);
    if (L > 1e-6) site[i] = [acc[i * 3] / L, acc[i * 3 + 1] / L, acc[i * 3 + 2] / L];
  }
}
site.sort((a, b) => b[1] - a[1]); // the shader looks sites up by a band around y
const pts = new Float32Array(TEX * 4);
for (let i = 0; i < NPTS; i++) {
  pts[i * 4] = site[i][0];
  pts[i * 4 + 1] = site[i][1];
  pts[i * 4 + 2] = site[i][2];
  pts[i * 4 + 3] = 1;
}
const ptsTex = new THREE.DataTexture(pts, TEX, 1, THREE.RGBAFormat, THREE.FloatType);
ptsTex.minFilter = ptsTex.magFilter = THREE.NearestFilter;
ptsTex.needsUpdate = true;
// hexagon circumradius on the unit sphere for N sites (area 4π/N ≈ 2.598 a²)
const CELL_A = Math.sqrt(4.836 / NPTS);

const CELL = `
  uniform sampler2D uPts; uniform float uN, uA;
  // nearest + second-nearest site around N (sites are sorted by y, so only a
  // band of indices can be close)
  void cellLookup(vec3 nrm, out vec3 c1, out float ji, out float edge){
    int ic = int(floor((1.0 - nrm.y) * uN * 0.5));
    float d1 = 9.0, d2 = 9.0; ji = 0.0; c1 = vec3(0.0,1.0,0.0);
    for (int k = -52; k <= 52; k++){
      int idx = ic + k;
      if (idx < 0 || idx >= int(uN)) continue;
      vec3 q = texture2D(uPts, vec2((float(idx)+0.5)/${TEX}.0, 0.5)).xyz;
      float d = distance(q, nrm);
      if (d < d1){ d2 = d1; d1 = d; ji = float(idx); c1 = q; }
      else if (d < d2) d2 = d;
    }
    edge = (d2 - d1) / uA; // 0 on a cell boundary, ≈1 at the centre
  }
`;

function apply(t) {
  const L = P.layers;

  // appear: easeOutBack scale, then a steady hold, then the fade-out
  const a = clamp01(t / 0.55);
  const s = 1 + 2.4 * Math.pow(a - 1, 3) + 1.4 * Math.pow(a - 1, 2); // overshoot ≈1.06
  const scale = 0.22 + 0.78 * clamp01(s);
  dome.scale.setScalar(scale);
  bandPivot.scale.setScalar(scale);

  const domeI = env(t, [[0, 0], [0.22, 0.9], [0.7, 1.0], [T_FADE, 1.0], [4.35, 0.55], [4.9, 0]]) * (L.dome ? 1 : 0);
  const reveal = env(t, [[0.08, 1.25], [0.92, -1.25]]);
  const dis = Math.max(0, t - T_FADE);

  // band: tilted ≈30°, ≈0.12 rev/s about Y while the shield holds
  const phi = 2.15 + 2 * Math.PI * 0.12 * (Math.min(Math.max(t, T_HOLD), T_FADE) - T_HOLD);
  bandPivot.rotation.y = phi;
  axis.set(Math.sin(TILT) * Math.cos(phi), Math.cos(TILT), -Math.sin(TILT) * Math.sin(phi));
  const bandI = env(t, [[0.5, 0], [0.95, 1.0], [T_FADE, 1.0], [4.3, 0.45], [4.8, 0]]) * (L.band ? 1 : 0);

  for (let i = 0; i < 2; i++) {
    const dt = t - RIP_T[i];
    const k = dt >= 0 && dt < 0.62 ? dt / 0.62 : -1;
    if (k < 0 || !L.ripples) ripV[i].set(0, 1, 0, 0);
    else {
      const hh = h11(i * 3.7 + 1.3);
      const hy = 0.15 + 0.7 * h11(i * 9.1 + 4.4);
      const rr = Math.sqrt(Math.max(0, 1 - hy * hy));
      const th = hh * Math.PI * 2;
      ripV[i].set(Math.cos(th) * rr, hy, Math.sin(th) * rr, k * 2.3 * (1 - 0.35 * k));
    }
  }

  const D = domeMat.uniforms;
  D.uTime.value = t;
  D.uI.value = domeI;
  D.uHex.value = L.hex ? 1 : 0;
  D.uRim.value = L.rim ? 1 : 0;
  D.uReveal.value = reveal;
  D.uDis.value = dis;
  D.uAxis.value.copy(axis);
  D.uBandI.value = bandI;
  D.uFloor.value = CY - R * scale;
  D.uRip.value = ripV;
  dome.visible = domeI > 1e-3;

  bandMat.uniforms.uTime.value = t;
  bandMat.uniforms.uI.value = bandI;
  bandMat.uniforms.uSweep.value = env(t, [[0.55, 0], [1.05, 1.2]]);
  haloMat.uniforms.uI.value = bandI * env(t, [[0.55, 0], [1.05, 1]]);
  bandPivot.visible = bandI > 1e-3;

  ringMat.uniforms.uI.value = bandI * 0.85;
  ring.visible = bandI > 1e-3;
  poolMat.uniforms.uI.value = domeI * 0.9;
  pool.visible = domeI > 1e-3;

  ground.material.uniforms.uPool.value = domeI * 0.9;
  ground.material.uniforms.uGold.value = bandI * 0.9;

  shieldLight.intensity = domeI * 2.6;
  marker.visible = !!L.marker;

  sparkMat.uniforms.uT.value = t;
  sparkMat.uniforms.uI.value = (L.dome ? 1 : 0) * env(t, [[0, 1], [0.85, 1], [1.05, 0]]);
  sparks.visible = t < 1.05 && !!L.dome;

  bloom.enabled = L.post;
}

let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1000);
  last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom;
  bloom.threshold = P.threshold;
  apply(P.time);
  composer.render();
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

window.spikeSetTime = (t) => {
  P.play = false;
  P.time = t;
  apply(t);
};
window.__spikeReady = true;
