const clamp01 = (x) => (x < 0 ? 0 : x > 1 ? 1 : x);
const ss = (a, b, x) => {
  const t = clamp01((x - a) / (b - a));
  return t * t * (3 - 2 * t);
};
// piecewise-linear envelope over [t,v] pairs
const env = (t, pts) => {
  if (t <= pts[0][0]) return pts[0][1];
  for (let i = 1; i < pts.length; i++) {
    if (t <= pts[i][0]) {
      const [t0, v0] = pts[i - 1];
      const [t1, v1] = pts[i];
      return v0 + (v1 - v0) * ss(0, 1, (t - t0) / (t1 - t0));
    }
  }
  return pts[pts.length - 1][1];
};
// deterministic hash: the only source of per-instance variation
const h11 = (n) => {
  let p = (n * 0.1031) % 1;
  p *= p + 33.33;
  p *= p + p;
  return p - Math.floor(p);
};

// beam length is the one piece of shared state every layer keys off
function beamLen(t) {
  if (t < T_EXT0) return 0;
  if (t < T_EXT1) {
    const a = (t - T_EXT0) / (T_EXT1 - T_EXT0);
    return BEAM_L * (1 - Math.pow(2, -10 * a)); // easeOutExpo: snaps out in ~2 frames
  }
  if (t < T_OFF0) return BEAM_L;
  if (t < T_OFF1) return BEAM_L * (1 - ss(T_OFF0, T_OFF1, t)); // retracts from the far end
  return 0;
}

// unit cylinder along +X, centred; uv.y = 1 at the emitter end
function beamGeo(seg) {
  const g = new THREE.CylinderGeometry(1, 1, 1, seg, 1, true);
  g.rotateZ(-Math.PI / 2);
  return g;
}
const BEAM_VS = `
  varying vec2 vUv; varying vec3 vN; varying vec3 vV;
  void main(){
    vUv = uv;
    vN = normalize(normalMatrix * normal);
    vec4 mv = modelViewMatrix * vec4(position,1.0);
    vV = -mv.xyz;
    gl_Position = projectionMatrix * mv;
  }`;

// core: hard white centre, magenta silhouette by fresnel, low-contrast fast stripe
const coreMat = new THREE.ShaderMaterial({
  uniforms: { uTime: { value: 0 }, uI: { value: 0 }, uLen: { value: 0 }, uThin: { value: 0 } },
  vertexShader: BEAM_VS,
  fragmentShader: `${GLSL}
    varying vec2 vUv; varying vec3 vN; varying vec3 vV;
    uniform float uTime, uI, uLen, uThin;
    void main(){
      float f = 1.0 - abs(dot(normalize(vN), normalize(vV)));
      float edge = pow(clamp(f,0.0,1.0), 1.5);
      float d = (1.0 - vUv.y) * uLen;                 // world distance from the emitter
      float stripe = fract(d*14.0 - uTime*6.0);
      float sm = 0.88 + 0.22*ssm(0.35,0.55,stripe);   // low contrast: the core stays solid
      vec3 col = mix(vec3(1.62,1.54,1.68), vec3(1.55,0.05,1.05), pow(edge,0.5));
      float tip = ssm(0.0, 0.22, uLen - d);           // soften only the leading tip
      float root = 0.75 + 0.25*ssm(0.0,0.5,d);
      float thin = mix(1.0, 1.0-edge*0.85, uThin);    // shut-off: collapses to a hairline
      gl_FragColor = vec4(col * sm * tip * root * thin * uI, 1.0);
    }`,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
  side: THREE.FrontSide,
});
const core = new THREE.Mesh(beamGeo(24), coreMat);
scene.add(core);

// sheath: magenta body with two panning stripe frequencies at a phase offset,
// a fresnel rim, and a hashed per-step edge flicker
function sheathMat(cfg) {
  return new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uI: { value: 0 },
      uLen: { value: 0 },
      uFlick: { value: 0 },
      uStreak: { value: cfg.streak },
      uRim: { value: cfg.rim },
      uSoft: { value: cfg.soft },
      uCol: { value: new THREE.Color(cfg.col) },
    },
    vertexShader: BEAM_VS,
    fragmentShader: `${GLSL}
      varying vec2 vUv; varying vec3 vN; varying vec3 vV;
      uniform float uTime, uI, uLen, uFlick, uStreak, uRim, uSoft;
      uniform vec3 uCol;
      void main(){
        float f = 1.0 - abs(dot(normalize(vN), normalize(vV)));
        float d = (1.0 - vUv.y) * uLen;
        // streak lines are per-ring: each circumferential band gets its own phase
        float ring = floor(vUv.x*15.0);
        float o = h11(ring*3.17+1.7);
        float s1 = fract(d*2.6 - uTime*3.4 + o);
        float s2 = fract(d*6.1 - uTime*6.6 + o*2.3 + 0.37);
        float streak = ssm(0.70,0.745,s1)*1.35 + ssm(0.855,0.885,s2)*0.95;
        float body = mix(1.0, 0.08 + streak*0.9, uStreak);
        float rim = pow(clamp(f,0.0,1.0), 3.0) * uRim;
        float soft = mix(1.0, 1.0 - pow(clamp(f,0.0,1.0),1.4), 0.0) ;
        float tip = ssm(0.0, 0.7, uLen - d);
        // edge flicker holds for 1/10 s at a time
        float fl = 0.82 + 0.34*h11(uFlick + ring*0.91);
        float a = (body*fl + rim) * tip * uI * uSoft;
        gl_FragColor = vec4(uCol * a, 1.0);
      }`,
    transparent: true,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
    side: THREE.FrontSide,
  });
}
const shMat = sheathMat({ col: 0xff0ad0, streak: 1.0, rim: 2.2, soft: 0.22 });
const sheath = new THREE.Mesh(beamGeo(30), shMat);
scene.add(sheath);

  uniforms: { uI: { value: 0 }, uLen: { value: 0 }, uH: { value: BAND_QH }, uCx: { value: 0 } },
  vertexShader: `
    uniform float uLen, uH, uCx; varying vec2 vUv;

      vUv = uv;
      vec4 mv = modelViewMatrix * vec4(uCx, ${EY.toFixed(3)}, 0.0, 1.0);

  fragmentShader: `${GLSL}
    varying vec2 vUv; uniform float uI, uH;
    void main(){
      float y = abs(vUv.y-0.5)*2.0*uH;                 // world units from the axis
      float e = ${EDGE.toFixed(4)};                    // 8% of the band height
      float body  = 1.0 - ssm(${BODY_H.toFixed(3)}-e, ${BODY_H.toFixed(3)}+e, y);
      float inner = 1.0 - ssm(${INNER_H.toFixed(3)}-e, ${INNER_H.toFixed(3)}+e, y);
      float outer = 1.0 - ssm(${OUTER_H.toFixed(3)}-e*1.6, ${OUTER_H.toFixed(3)}+e*1.6, y);
      float halo  = exp(-y*y*1.6) * 0.10;              // faint seat, so it is not pasted on
      vec3 c = mix(vec3(0.34,0.010,0.72), vec3(0.78,0.015,0.90), body);
      c = mix(c, vec3(1.25,0.134,1.09), inner);
      float a = outer*0.34 + body*0.56 + halo;
      a *= ssm(0.0, 0.16, vUv.x) * (0.72 + 0.28*vUv.x);  // thins toward the far tip
      gl_FragColor = vec4(c*a*uI, 1.0);
    }`,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
});

  g.instanceCount = N_RIB;

const ribMat = new THREE.ShaderMaterial({
  uniforms: { uTime: { value: 0 }, uI: { value: 0 }, uLen: { value: 0 } },
  vertexShader: `${GLSL}
    attribute float aSeed; uniform float uTime, uI, uLen;
    varying vec2 vUv; varying vec3 vC; varying float vK;
    void main(){
      vUv = uv;
      float k  = floor(uTime*6.0) + aSeed;
      float L  = 0.8 + 1.4*h11(k*1.31);                 // 0.8 .. 2.2 world units
      float th = 0.04 + 0.04*h11(k*3.11+1.0);           // 0.04 .. 0.08 tall
      float sp = 6.0 + 4.0*h11(k*5.33+2.0);             // 6 .. 10 units/s toward -x
      float hy = (h11(k*7.91+3.0)*2.0-1.0);
      hy = sign(hy) * (0.05 + 0.36*pow(abs(hy),0.8));    // a few clear the body edge
      float skew = (h11(k*19.3+6.0)*2.0-1.0) * 0.35;
      float x0 = ${EX.toFixed(3)} - fract(h11(k*11.3+4.0) + uTime*sp/${BEAM_L.toFixed(2)})*(uLen+L) + L;
      float ox = (x0 - position.x*L) - ${EX.toFixed(3)} + position.y*skew;
      vec4 mv = modelViewMatrix * vec4(${EX.toFixed(3)}, ${EY.toFixed(3)}, 0.0, 1.0);
      mv.xy += vec2(ox, hy + position.y*th*2.0);
      vK = 0.65 + 0.35*h11(k*23.7+8.0);
      vC = h11(k*13.7+5.0) > 0.50 ? vec3(2.55,1.15,2.35) : vec3(2.05,0.38,1.85);
      gl_Position = projectionMatrix * mv;
    }`,
  fragmentShader: `${GLSL}
    varying vec2 vUv; varying vec3 vC; varying float vK; uniform float uI;
    void main(){
      float y = abs(vUv.y-0.5)*2.0;
      float a = 1.0 - ssm(0.80, 1.0, y);                 // hard slab edge
      a *= ssm(0.0,0.07,vUv.x) * (1.0-ssm(0.88,1.0,vUv.x));
      gl_FragColor = vec4(vC*a*vK*uI, 1.0);
    }`,
  transparent: true,
  blending: THREE.AdditiveBlending,
  depthWrite: false,
});

// ---------------------------------------------------------------- flame tongues
// flat cel-style licks in two palettes: dark violet behind the body, light lilac
// in front. Screen-aligned, and the whole set swaps on a 10 Hz flipbook step.
function tongueGeo(n, salt) {
  const NS = 16;
  const pos = [];
  const idx = [];
  for (let i = 0; i <= NS; i++) {
    const s = i / NS;
    pos.push(s, -1, 0, s, 1, 0);
  }
  for (let i = 0; i < NS; i++) {
    const a = i * 2;
    idx.push(a, a + 1, a + 2, a + 1, a + 3, a + 2);
  }
  const g = new THREE.InstancedBufferGeometry();
  g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  g.setIndex(idx);
  const seed = new Float32Array(n);
  const side = new Float32Array(n);
  const u0 = new Float32Array(n);
  for (let i = 0; i < n; i++) {
    seed[i] = i * 7.13 + salt;
    side[i] = i % 2 === 0 ? 1 : -1;
    // the last two of each set are short licks pinned at the far end
    u0[i] = i >= n - 2 ? 0.86 + 0.11 * h11(i * 3.3 + salt) : 0.04 + 0.74 * h11(i * 2.7 + salt);
  }
  g.setAttribute("aSeed", new THREE.InstancedBufferAttribute(seed, 1));
  g.setAttribute("aSide", new THREE.InstancedBufferAttribute(side, 1));
  g.setAttribute("aU0", new THREE.InstancedBufferAttribute(u0, 1));
  g.instanceCount = n;
  return g;
}
const TONGUE_VS = `${GLSL}
  attribute float aSeed, aSide, aU0;
  uniform float uTime, uLen, uSalt;
  varying float vS; varying float vFade;
  void main(){
    float s = position.x, v = position.y;
    float k = floor(uTime*10.0) + aSeed + uSalt;   // flipbook step: shape holds 1/10 s
    float L = 0.8 + 1.6*h11(k*1.31);               // 0.8 .. 2.4 long
    float W = 0.15 + 0.30*h11(k*3.77+2.1);         // 0.15 .. 0.45 thick
    float wob = h11(k*5.51+7.7);
    // some licks ride over the body itself, not only outside it
    float yc = 0.02 + 0.62*h11(k*9.13+3.3);
    float bx = ${EX.toFixed(3)} - aU0*uLen;
    float ox = -s*L;                                // tips trail forward (-x)
    float w = W*0.5 * pow(s+0.05, 0.30) * pow(max(1.0-s,0.0), 0.85);
    float cy = 0.16*sin(s*3.1 + wob*6.2831)*s;
    float oy = aSide*(yc + cy) + v*w;
    vec4 mv = modelViewMatrix * vec4(bx, ${EY.toFixed(3)}, 0.0, 1.0);
    mv.xy += vec2(ox, oy);
    vS = s;
    vFade = ssm(${(EX - BEAM_L).toFixed(3)} - 0.75, ${(EX - BEAM_L).toFixed(3)} - 0.05, bx + ox);
    gl_Position = projectionMatrix * mv;
  }`;
function tongueMat(col, alpha, salt) {
  return new THREE.ShaderMaterial({
    uniforms: {
      uTime: { value: 0 },
      uLen: { value: 0 },
      uI: { value: 0 },
      uSalt: { value: salt },
      uCol: { value: new THREE.Color(col) },
      uA: { value: alpha },
    },
    vertexShader: TONGUE_VS,
    fragmentShader: `
      varying float vS; varying float vFade;
      uniform vec3 uCol; uniform float uI, uA;
      void main(){
        float a = uI * uA * vFade;
        if (a <= 0.001) discard;
        gl_FragColor = vec4(uCol, a);     // flat fill: no gradient, cel style
      }`,
    transparent: true,
    depthWrite: false,
    side: THREE.DoubleSide,
  });
}
const tongueDark = new THREE.Mesh(tongueGeo(N_TONGUE_D, 1.9), tongueMat(0x5a1c8a, 0.48, 0));
const tongueLight = new THREE.Mesh(tongueGeo(N_TONGUE_L, 5.3), tongueMat(0xbfaad6, 0.50, 37));
tongueDark.frustumCulled = tongueLight.frustumCulled = false;

tongueLight.renderOrder = 6;  // in front of it

// ---------------------------------------------------------------- emitter flare
// screen-aligned quads at the emitter: a tall lens streak, a radial muzzle burst
// and a hot core sprite.
function billboard(fs, uniforms, extraVs = "") {
  return new THREE.ShaderMaterial({
    uniforms: Object.assign({ uTime: { value: 0 }, uI: { value: 0 } }, uniforms),
    vertexShader: `${GLSL}
      uniform float uTime, uI; uniform vec2 uSize; ${extraVs}
      varying vec2 vUv;
      void main(){
        vUv = uv;
        vec4 mv = modelViewMatrix * vec4(0.0,0.0,0.0,1.0);
        mv.xy += position.xy * uSize;
        gl_Position = projectionMatrix * mv;
      }`,
    fragmentShader: fs,
    transparent: true,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
  });
}
const flareMat = billboard(
  `${GLSL}
   varying vec2 vUv; uniform float uTime, uI;
   void main(){
     vec2 p = vUv - 0.5;
     float x = abs(p.x)*2.0, y = abs(p.y)*2.0;
     float core = exp(-x*x*64.0)*2.0;    // white blade ≈ 0.10 wide
     float wide = exp(-x*x*6.4)*0.85;    // magenta body ≈ 0.32 wide
     float halo = exp(-x*x*1.6)*0.40;    // saturated magenta halo
     float v = pow(max(0.0, 1.0-y), 1.2) * (0.18 + 0.82*exp(-y*y*3.0));
     float a = (core + wide + halo*exp(-y*y*7.0)) * v;
     vec3 col = mix(vec3(1.9,0.06,1.7), vec3(3.0,2.7,3.0), clamp(core,0.0,1.0)*(1.0-y*0.5));
     gl_FragColor = vec4(col*a*uI, 1.0);
   }`,
  { uSize: { value: new THREE.Vector2(0.82, 3.2) } }
);
const flare = new THREE.Mesh(quad, flareMat);

// ---------------------------------------------------------------- point layers
// every dot layer is a THREE.Points whose vertex shader is closed-form in t
function points(n, vs, fs, uniforms) {
  const g = new THREE.BufferGeometry();
  const pos = new Float32Array(n * 3);
  const seed = new Float32Array(n);
  for (let i = 0; i < n; i++) seed[i] = i;
  g.setAttribute("position", new THREE.BufferAttribute(pos, 3));
  g.setAttribute("aSeed", new THREE.BufferAttribute(seed, 1));
  g.boundingSphere = new THREE.Sphere(new THREE.Vector3(0, EY, 0), 40);
  const m = new THREE.ShaderMaterial({
    uniforms: Object.assign({ uTime: { value: 0 }, uI: { value: 0 }, uLen: { value: 0 } }, uniforms),
    vertexShader: `${GLSL}
      attribute float aSeed; uniform float uTime, uI, uLen;
      varying float vA; varying vec3 vC;
      ${vs}`,
    fragmentShader: `
      varying float vA; varying vec3 vC;
      void main(){
        float r = length(gl_PointCoord-0.5)*2.0;
        float a = exp(-r*r*5.0);
        if (vA <= 0.001) discard;
        gl_FragColor = vec4(vC*a*vA, 1.0);
      }`,
    transparent: true,
    blending: THREE.AdditiveBlending,
    depthWrite: false,
  });
  const p = new THREE.Points(g, m);
  p.frustumCulled = false;
  scene.add(p);
  return p;
}

// ---------------------------------------------------------------- per-frame state
function apply(t) {
  const L = P.layers;
  const len = beamLen(t);
  const on = len > 1e-3;

  // beam meshes: scaled unit cylinders, right end pinned at the emitter
  for (const [m, r] of [[core, CORE_R], [sheath, SHEATH_R]]) {
    m.scale.set(Math.max(len, 1e-4), r, r);
    m.position.set(EX - len * 0.5, EY, 0);
  }

  const beamI = env(t, [[T_EXT0, 0], [T_EXT1, 1], [T_OFF0, 1], [T_OFF0 + 0.10, 0.28], [T_OFF1, 0]]);
  const pulse = 0.92 + 0.08 * Math.sin(t * 17.0) + 0.06 * h11(Math.floor(t * 13));
  const thin = ss(T_OFF0, T_OFF0 + 0.12, t);

  coreMat.uniforms.uTime.value = t;
  coreMat.uniforms.uLen.value = len;
  coreMat.uniforms.uThin.value = thin;
  coreMat.uniforms.uI.value = beamI * pulse * 1.15 * (L.core ? 1 : 0);
  core.visible = on && L.core;

  const shI = beamI * (0.9 + 0.2 * h11(Math.floor(t * 10) + 5)) * (L.sheath ? 1 : 0);
  shMat.uniforms.uTime.value = t;
  shMat.uniforms.uLen.value = len;
  shMat.uniforms.uFlick.value = Math.floor(t * 10);
  shMat.uniforms.uI.value = shI * 0.55;
  bandMat.uniforms.uLen.value = len;
  bandMat.uniforms.uCx.value = EX - len * 0.5;
  bandMat.uniforms.uI.value = shI * 1.0;
  ribMat.uniforms.uTime.value = t;
  ribMat.uniforms.uLen.value = len;
  ribMat.uniforms.uI.value = shI * 1.15;
  sheath.visible = band.visible = ribbons.visible = on && L.sheath;

  const tongueI = env(t, [[T_EXT1 - 0.05, 0], [T_EXT1 + 0.15, 1], [T_OFF0 - 0.05, 1], [T_OFF0 + 0.08, 0]]);
  for (const m of [tongueDark.material, tongueLight.material]) {
    m.uniforms.uTime.value = t;
    m.uniforms.uLen.value = len;
    m.uniforms.uI.value = tongueI * (L.tongues ? 1 : 0);
  }
  tongueDark.visible = tongueLight.visible = tongueI > 1e-3 && L.tongues;

  // flare pulses through the hold; a white flash punches at ignition
  const flash = env(t, [[T_EXT0 - 0.06, 0], [T_EXT0 + 0.02, 1], [T_EXT0 + 0.28, 0]]);
  const flareI =
    (env(t, [[T_EXT0, 0], [T_EXT1, 1.0], [T_OFF0, 1.0], [T_OFF1 + 0.1, 0]]) *
      (0.82 + 0.18 * Math.sin(t * 9.3) + 0.1 * h11(Math.floor(t * 12) + 3)) +
      flash * 1.4) *
    (L.flare ? 1 : 0);
  flareMat.uniforms.uI.value = flareI;
  flareMat.uniforms.uTime.value = t;
  rayMat.uniforms.uI.value = flareI * 1.6;
  rayMat.uniforms.uTime.value = t;
  hotMat.uniforms.uI.value = flareI * 0.95;
  // charge ball: 0.05 -> 0.25 radius over the gather, then a white punch at 1.0
  const chgA = clamp01(t / T_EXT0);
  const ballR = 0.06 + 0.22 * Math.pow(chgA, 1.5);
  ballMat.uniforms.uSize.value.set(ballR * 7.0, ballR * 7.0);
  ballMat.uniforms.uI.value =
    (L.charge ? 1 : 0) * (env(t, [[0.02, 0], [0.22, 0.75], [T_EXT0, 1.25], [T_EXT0 + 0.18, 0]]) + flash * 0.8);
  ball.visible = ballMat.uniforms.uI.value > 1e-3;
  flare.visible = rays.visible = flareI > 1e-3;
  hot.visible = hotMat.uniforms.uI.value > 1e-3;

  chg.material.uniforms.uTime.value = t;
  chg.material.uniforms.uI.value = (L.charge ? 1 : 0) * (t < T_EXT0 + 0.06 ? 1 : 0);
  chg.visible = chg.material.uniforms.uI.value > 1e-3;

  spk.material.uniforms.uTime.value = t;
  spk.material.uniforms.uLen.value = len;
  spk.material.uniforms.uI.value = (L.sparks ? 1 : 0) * env(t, [[T_EXT1, 0], [T_EXT1 + 0.1, 1], [T_OFF0, 1], [T_OFF0 + 0.06, 0]]);
  spk.visible = spk.material.uniforms.uI.value > 1e-3;

  run.material.uniforms.uTime.value = t;
  run.material.uniforms.uI.value = (L.sparks ? 1 : 0) * env(t, [[T_OFF0, 0], [T_OFF0 + 0.05, 1], [3.95, 1], [4.15, 0]]);
  run.visible = run.material.uniforms.uI.value > 1e-3;

  res.material.uniforms.uTime.value = t;
  res.material.uniforms.uI.value = L.residual ? 1 : 0;
  res.visible = t >= T_RES0 && t <= T_RES1 + 0.1 && L.residual;

  const gu = ground.material.uniforms;
  gu.uLen.value = len;
  gu.uPool.value = beamI * 0.55 + flash * 0.35;
  gu.uEmit.value = flareI * 0.22 + env(t, [[0.3, 0], [T_EXT0, 0.10]]) * (L.charge ? 1 : 0);
  gu.uRes.value = res.visible ? env(t, [[T_RES0, 0], [3.9, 1], [4.6, 0.5], [T_RES1, 0]]) : 0;

  bloom.enabled = L.post;
}

let last = performance.now();
function frame(now) {
  const dt = Math.min(0.1, (now - last) / 1000);
  last = now;
  if (P.play) P.time = (P.time + dt * P.speed) % DUR;
  renderer.toneMappingExposure = P.exposure;
  bloom.strength = P.bloom;
  bloom.threshold = P.threshold;
  apply(P.time);
  composer.render();
  window.__spikeFrame = (window.__spikeFrame || 0) + 1;
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);

window.spikeSetTime = (t) => {
  P.play = false;
  P.time = t;
  apply(t);
};
window.__spikeReady = true;
